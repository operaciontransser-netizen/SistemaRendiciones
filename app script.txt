const VERSION_CODIGO = "RENDICIONES_9_6_4_3_2026-08-26";

// ID explícito de la planilla RENDICION OLD TRANSSER.
// Evita que la aplicación web lea otra planilla activa.
const ID_SPREADSHEET =
  "1v34bKh65YvbSHfAE0qfIDbFy9mArUxDoBkTsRUz6gOE";


function obtenerLibroRendiciones() {
  return SpreadsheetApp.openById(
    ID_SPREADSHEET
  );
}

let CALLBACK_JSONP = "";


function doGet(e) {
  reiniciarCachesConsulta();
  const libro = obtenerLibroRendiciones();

  const hojaRespuestas = libro.getSheetByName("RESPUESTAS");
  const hojaColaboradores = libro.getSheetByName("COLABORADORES");
  const hojaConsolidado = libro.getSheetByName("Consolidado");
  const hojaAbonos = libro.getSheetByName("ABONOS");
  const hojaEmpresas = libro.getSheetByName("EMPRESAS");
  const hojaComprobantesRechazados = libro.getSheetByName(
    "COMPROBANTES_RECHAZADOS"
  );
  const hojaComprobantesDuplicados = libro.getSheetByName(
    "COMPROBANTES_DUPLICADOS"
  );

  if (!hojaRespuestas) {
    return responderJSON({
      error: 'No existe la hoja "RESPUESTAS".'
    });
  }

  const parametros =
    e && e.parameter
      ? e.parameter
      : {};

  CALLBACK_JSONP = String(
    parametros.callback || ""
  ).trim();

  const foto = String(
    parametros.foto || ""
  ).trim();

  const id = String(
    parametros.id || ""
  ).trim();

  const emailUsuario = String(
    parametros.usuario || ""
  )
    .trim()
    .toLowerCase();

  const resumen = String(
    parametros.resumen || ""
  ).trim();

  const accion = String(
    parametros.accion || ""
  ).trim().toLowerCase();

  const empresaSeleccionada = String(
    parametros.empresa || ""
  ).trim();

  const solicitarColaboradores = String(
    parametros.colaboradores || ""
  ).trim();

  const solicitarEmpresas = String(
    parametros.empresas || ""
  ).trim();

  const solicitarComprobantesRechazados = String(
    parametros.comprobantes_rechazados || ""
  ).trim();

  const solicitarAbonos = String(
    parametros.abonos || ""
  ).trim();

  const solicitarCartola = String(
    parametros.cartola || ""
  ).trim();

  // ======================================================
  // FOTOGRAFÍA PARA PDF
  // ======================================================

  if (foto) {
    return obtenerFotoDriveComoBase64(
      hojaRespuestas,
      foto
    );
  }

  // ======================================================
  // PERMISOS DEL USUARIO
  // ======================================================

  let permisos = null;

  if (emailUsuario) {
    permisos = obtenerPermisosUsuario(
      emailUsuario
    );

    if (
      !permisos ||
      !permisos.encontrado
    ) {
      return responderJSON({
        error: "Usuario no autorizado.",
        usuario: emailUsuario
      });
    }

    if (!permisos.activo) {
      return responderJSON({
        error: "El usuario se encuentra inactivo.",
        usuario: emailUsuario
      });
    }
  }

  // ======================================================
  // DIAGNÓSTICO / VERSIÓN (NO AFECTA EL USO NORMAL)
  // ======================================================

  if (parametros.version === "1") {
    return responderJSON({
      version: VERSION_CODIGO
    });
  }

  if (parametros.debug === "1") {
    return responderJSON({
      version: VERSION_CODIGO,

      spreadsheet_nombre:
        libro.getName(),

      spreadsheet_id:
        libro.getId(),

      hoja_respuestas:
        hojaRespuestas.getName(),

      ultima_fila_respuestas:
        hojaRespuestas.getLastRow(),

      ultima_columna_respuestas:
        hojaRespuestas.getLastColumn(),

      diagnostico_fechas:
        obtenerDiagnosticoFechas(
          hojaRespuestas
        ),

      usuario_recibido: emailUsuario,
      id_recibido: id,
      permisos: permisos
    });
  }

  if (
    !permisos ||
    !permisos.encontrado ||
    !permisos.activo
  ) {
    return responderJSON({
      error:
        "Debe indicar un usuario activo y autorizado."
    });
  }

  // ======================================================
  // MONITOR OPERATIVO (SOLO SUPER ADMIN)
  // ======================================================

  if (accion === "monitor_sistema") {
    return obtenerMonitorSistema(
      libro,
      hojaRespuestas,
      hojaComprobantesDuplicados,
      permisos
    );
  }

  // ======================================================
  // FONDOS Y ABONOS
  // ======================================================

  if (
    accion === "registrar_abono"
  ) {
    return registrarAbonoDesdeGet(
      hojaAbonos,
      hojaColaboradores,
      parametros,
      permisos,
      empresaSeleccionada
    );
  }

  if (
    accion === "anular_abono"
  ) {
    return anularAbonoDesdeGet(
      hojaAbonos,
      hojaColaboradores,
      parametros,
      permisos,
      empresaSeleccionada
    );
  }

  if (
    solicitarAbonos === "1" ||
    accion === "listar_abonos"
  ) {
    return obtenerListadoAbonos(
      hojaAbonos,
      hojaColaboradores,
      permisos,
      empresaSeleccionada,
      parametros
    );
  }

  if (
    solicitarCartola === "1" ||
    accion === "obtener_cartola"
  ) {
    return obtenerCartolaColaborador(
      hojaAbonos,
      hojaRespuestas,
      hojaColaboradores,
      permisos,
      empresaSeleccionada,
      parametros
    );
  }

  // ======================================================
  // ELIMINAR RENDICIÓN (SOLO SUPER ADMIN)
  // ======================================================

  if (accion === "eliminarrendicion") {
    return eliminarRendicionDesdeGet(
      hojaRespuestas,
      parametros,
      permisos
    );
  }


  // ======================================================
  // AUTORIZAR RENDICIÓN
  // ======================================================

  if (accion === "autorizar_rendicion") {
    return procesarAutorizacionDesdeGet(
      hojaRespuestas,
      parametros,
      permisos
    );
  }


  // ======================================================
  // EMPRESAS DISPONIBLES
  // ======================================================

  if (solicitarEmpresas === "1") {
    return obtenerListadoEmpresas(
      hojaEmpresas,
      permisos
    );
  }

  // ======================================================
  // REPORTE DE COMPROBANTES RECHAZADOS
  // ======================================================

  if (solicitarComprobantesRechazados === "1") {
    return obtenerListadoComprobantesRechazados(
      hojaComprobantesRechazados,
      permisos,
      empresaSeleccionada,
      parametros
    );
  }

  if (
    accion ===
    "finalizar_revision_comprobantes"
  ) {
    return procesarRevisionComprobantesDesdeGet(
      hojaRespuestas,
      hojaColaboradores,
      parametros,
      permisos,
      empresaSeleccionada
    );
  }

  // ======================================================
  // LISTADO DE COLABORADORES
  // ======================================================

  if (solicitarColaboradores === "1") {
    return obtenerListadoColaboradores(
      hojaColaboradores,
      hojaAbonos,
      hojaRespuestas,
      permisos,
      empresaSeleccionada
    );
  }


  // ======================================================
  // RESUMEN FINANCIERO PARA DASHBOARD
  // ======================================================

  if (resumen === "1") {
    return obtenerResumenFinanciero(
      hojaAbonos,
      hojaRespuestas,
      hojaColaboradores,
      permisos,
      empresaSeleccionada
    );
  }


  // ======================================================
  // DETALLE DE UNA RENDICIÓN
  // ======================================================

  if (id) {
    return obtenerDetalleRendicion(
      hojaRespuestas,
      hojaColaboradores,
      hojaConsolidado,
      hojaComprobantesDuplicados,
      id,
      permisos,
      empresaSeleccionada
    );
  }

  // ======================================================
  // LISTADO DE RENDICIONES
  // ======================================================

  return obtenerListadoRendiciones(
    hojaRespuestas,
    permisos,
    empresaSeleccionada,
    hojaColaboradores,
    hojaComprobantesDuplicados
  );
}



// ======================================================
// DIAGNÓSTICO DE FECHAS
// ======================================================

function obtenerDiagnosticoFechas(
  hoja
) {
  const ultimaFila =
    hoja.getLastRow();

  const ultimaColumna =
    hoja.getLastColumn();

  const encabezados =
    hoja
      .getRange(
        1,
        1,
        1,
        ultimaColumna
      )
      .getDisplayValues()[0];

  const idxProcesado =
    buscarIndice(
      encabezados,
      "procesado_en"
    );

  const idxAlternativa =
    buscarIndiceCualquiera(
      encabezados,
      [
        "Submission Date",
        "submission_date",
        "submitted_at",
        "submittedAt",
        "Last Update Date",
        "last_update_date"
      ]
    );

  let muestras = [];

  if (ultimaFila >= 2) {
    const cantidad =
      Math.min(
        ultimaFila - 1,
        10
      );

    const datos =
      hoja
        .getRange(
          2,
          1,
          cantidad,
          ultimaColumna
        )
        .getDisplayValues();

    muestras =
      datos.map(
        (fila) => ({
          procesado_en:
            idxProcesado >= 0
              ? valorFila(
                  fila,
                  idxProcesado
                )
              : "",
          fecha_alternativa:
            idxAlternativa >= 0
              ? valorFila(
                  fila,
                  idxAlternativa
                )
              : ""
        })
      );
  }

  return {
    indice_procesado_en:
      idxProcesado,
    indice_fecha_alternativa:
      idxAlternativa,
    encabezado_procesado_en:
      idxProcesado >= 0
        ? encabezados[idxProcesado]
        : "",
    encabezado_fecha_alternativa:
      idxAlternativa >= 0
        ? encabezados[idxAlternativa]
        : "",
    muestras:
      muestras
  };
}


// ======================================================
// LISTADO DE EMPRESAS
// ======================================================

function obtenerListadoEmpresas(
  hojaEmpresas,
  permisos
) {
  const cacheKey = obtenerClaveCacheConsulta(
    "EMPRESAS",
    permisos,
    ""
  );
  const cacheado = leerCacheJSON(cacheKey);
  if (cacheado && Array.isArray(cacheado)) {
    return responderJSON(cacheado);
  }

  if (!hojaEmpresas) {
    return responderJSON({
      error:
        'No existe la hoja "EMPRESAS".'
    });
  }

  const datos = hojaEmpresas
    .getDataRange()
    .getDisplayValues();

  if (datos.length < 2) {
    return responderJSON([]);
  }

  const encabezados = datos[0];
  const idxCodigo = buscarIndice(
    encabezados,
    "codigo_empresa"
  );
  const idxNombre = buscarIndice(
    encabezados,
    "nombre_empresa"
  );
  const idxRut = buscarIndice(
    encabezados,
    "rut_empresa"
  );
  const idxLogo = buscarIndice(
    encabezados,
    "logo_url"
  );

  if (idxCodigo < 0 || idxNombre < 0) {
    return responderJSON({
      error:
        "EMPRESAS debe contener codigo_empresa y nombre_empresa."
    });
  }

  const empresaUsuario =
    normalizarEmpresaSistema(
      permisos && permisos.empresa
    );

  const resultado = datos
    .slice(1)
    .map((fila) => ({
      codigo_empresa: String(
        valorFila(fila, idxCodigo) || ""
      ).trim(),
      nombre_empresa: String(
        valorFila(fila, idxNombre) || ""
      ).trim(),
      rut_empresa: String(
        valorFila(fila, idxRut) || ""
      ).trim(),
      logo_url: String(
        valorFila(fila, idxLogo) || ""
      ).trim()
    }))
    .filter((empresa) => {
      if (!empresa.codigo_empresa) {
        return false;
      }

      return (
        esRolSuperAdmin(permisos) ||
        !empresaUsuario ||
        normalizarEmpresaSistema(
          empresa.codigo_empresa
        ) === empresaUsuario
      );
    });

  guardarCacheJSON(cacheKey, resultado, 60);
  return responderJSON(resultado);
}


// ======================================================
// LISTADO DE COLABORADORES
// ======================================================

function obtenerListadoColaboradores(
  hojaColaboradores,
  hojaAbonos,
  hojaRespuestas,
  permisos,
  empresaSeleccionada
) {
  const cacheKey = obtenerClaveCacheConsulta(
    "COLABORADORES",
    permisos,
    empresaSeleccionada
  );

  const cacheado = leerCacheJSON(cacheKey);
  if (cacheado && Array.isArray(cacheado)) {
    return responderJSON(cacheado);
  }

  if (
    !permisos ||
    !permisos.encontrado ||
    !permisos.activo
  ) {
    return responderJSON({
      error: "Usuario no autorizado."
    });
  }

  if (!hojaColaboradores) {
    return responderJSON({
      error:
        'No existe la hoja "COLABORADORES".'
    });
  }

  const esSuperAdmin =
    esRolSuperAdmin(permisos);

  const rutsPermitidos =
    Array.isArray(permisos.colaboradores)
      ? permisos.colaboradores.map(
          normalizarRutPermisos
        )
      : [];

  const datosColaboradores =
    hojaColaboradores
      .getDataRange()
      .getDisplayValues();

  if (datosColaboradores.length < 2) {
    return responderJSON([]);
  }

  const encabezadosColaboradores =
    datosColaboradores[0];

  const idxRutColaborador = buscarIndice(
    encabezadosColaboradores,
    "rut"
  );

  const idxNombreColaborador = buscarIndice(
    encabezadosColaboradores,
    "colaborador"
  );

  const idxCentroCosto = buscarIndice(
    encabezadosColaboradores,
    "centro de costo"
  );

  const idxEmpresaColaborador = buscarIndice(
    encabezadosColaboradores,
    "empresa"
  );

  const idxCorreoColaborador = buscarIndice(
    encabezadosColaboradores,
    "correo"
  );

  if (
    idxRutColaborador < 0 ||
    idxNombreColaborador < 0 ||
    idxCentroCosto < 0 ||
    idxEmpresaColaborador < 0 ||
    idxCorreoColaborador < 0
  ) {
    return responderJSON({
      error:
        "La hoja COLABORADORES no tiene todos los encabezados requeridos."
    });
  }

  const financierosPorRut =
    obtenerFinancierosDirectosPorRut(
      hojaAbonos,
      hojaRespuestas
    );

  const rendicionesPorRut =
    obtenerResumenRendicionesPorRut(
      hojaRespuestas
    );

  const resultado = [];

  for (
    let i = 1;
    i < datosColaboradores.length;
    i++
  ) {
    const fila =
      datosColaboradores[i];

    const rut = String(
      valorFila(
        fila,
        idxRutColaborador
      ) || ""
    ).trim();

    const rutNormalizado =
      normalizarRutPermisos(rut);

    if (!rutNormalizado) {
      continue;
    }

    if (
      !esSuperAdmin &&
      !rutsPermitidos.includes(
        rutNormalizado
      )
    ) {
      continue;
    }

    const empresaFila = String(
      valorFila(
        fila,
        idxEmpresaColaborador
      ) || ""
    ).trim();

    if (
      !empresaPermitidaParaConsulta(
        empresaFila,
        permisos,
        empresaSeleccionada
      )
    ) {
      continue;
    }

    const financiero =
      financierosPorRut[rutNormalizado] ||
      {
        monto_asignado: 0,
        monto_presentado: 0,
        monto_rendido: 0,
        monto_pendiente_revision: 0,
        monto_rechazado: 0,
        saldo: 0
      };

    const resumenRendiciones =
      rendicionesPorRut[rutNormalizado] ||
      {
        cantidad_rendiciones: 0,
        ultima_rendicion_id: "",
        ultima_rendicion_en: ""
      };

    resultado.push({
      rut: rut,
      colaborador: String(
        valorFila(
          fila,
          idxNombreColaborador
        ) || ""
      ).trim(),
      centro_costo: String(
        valorFila(
          fila,
          idxCentroCosto
        ) || ""
      ).trim(),
      empresa: empresaFila,
      correo: String(
        valorFila(
          fila,
          idxCorreoColaborador
        ) || ""
      ).trim(),
      monto_asignado:
        financiero.monto_asignado,
      monto_presentado:
        financiero.monto_presentado,
      monto_rendido:
        financiero.monto_rendido,
      monto_autorizado:
        financiero.monto_rendido,
      monto_pendiente_revision:
        financiero.monto_pendiente_revision,
      monto_rechazado:
        financiero.monto_rechazado,
      saldo:
        financiero.saldo,
      cantidad_rendiciones:
        resumenRendiciones.cantidad_rendiciones,
      ultima_rendicion_id:
        resumenRendiciones.ultima_rendicion_id,
      ultima_rendicion_en:
        resumenRendiciones.ultima_rendicion_en
    });
  }

  resultado.sort(
    (a, b) =>
      String(a.colaborador || "")
        .localeCompare(
          String(b.colaborador || ""),
          "es",
          { sensitivity: "base" }
        )
  );

  guardarCacheJSON(cacheKey, resultado, 30);
  return responderJSON(resultado);
}


function obtenerFinancierosDirectosPorRut(
  hojaAbonos,
  hojaRespuestas
) {
  const resultado = {};

  const asegurarRut = (rut) => {
    const clave = normalizarRutPermisos(rut);

    if (!clave) {
      return null;
    }

    if (!resultado[clave]) {
      resultado[clave] = {
        monto_asignado: 0,
        monto_presentado: 0,
        monto_rendido: 0,
        monto_pendiente_revision: 0,
        monto_rechazado: 0,
        saldo: 0
      };
    }

    return resultado[clave];
  };

  if (
    hojaAbonos &&
    hojaAbonos.getLastRow() >= 2
  ) {
    const datosAbonos = obtenerDatosHojaConsultaCached(
      hojaAbonos,
      "ABONOS"
    );
    const encabezados = datosAbonos[0];
    const idxColaborador = buscarIndice(
      encabezados,
      "COLABORADOR"
    );
    const idxDeposito = buscarIndice(
      encabezados,
      "DEPOSITO"
    );
    const idxEstado = buscarIndice(
      encabezados,
      "ESTADO"
    );

    if (idxColaborador >= 0 && idxDeposito >= 0) {
      datosAbonos.slice(1).forEach((fila) => {
        const estado =
          idxEstado >= 0
            ? String(
                valorFila(fila, idxEstado) ||
                "ACTIVO"
              ).trim().toUpperCase()
            : "ACTIVO";

        if (estado === "ANULADO") {
          return;
        }

        const item = asegurarRut(
          extraerRut(
            valorFila(fila, idxColaborador)
          )
        );

        if (item) {
          item.monto_asignado +=
            convertirNumero(
              valorFila(fila, idxDeposito)
            );
        }
      });
    }
  }

  if (
    hojaRespuestas &&
    hojaRespuestas.getLastRow() >= 2
  ) {
    const datosRespuestas = obtenerDatosHojaConsultaCached(
      hojaRespuestas,
      "RESPUESTAS"
    );
    const encabezados = datosRespuestas[0];
    const idxColaborador = buscarIndice(
      encabezados,
      "colaborador"
    );
    const idxRut = buscarIndice(
      encabezados,
      "rut"
    );
    const idxMonto = buscarIndiceCualquiera(
      encabezados,
      ["monto_total", "MONTO"]
    );
    const idxEstadoComprobante = buscarIndice(
      encabezados,
      "estado_comprobante"
    );
    const idxEstadoRendicion = buscarIndice(
      encabezados,
      "estado_rendicion"
    );

    if (idxColaborador >= 0 && idxMonto >= 0) {
      datosRespuestas.slice(1).forEach((fila) => {
        const colaborador = valorFila(
          fila,
          idxColaborador
        );
        const item = asegurarRut(
          extraerRut(colaborador) ||
          valorFila(fila, idxRut)
        );

        if (!item) {
          return;
        }

        const monto = convertirNumero(
          valorFila(fila, idxMonto)
        );
        item.monto_presentado += monto;

        const estadoComprobante =
          idxEstadoComprobante >= 0
            ? String(
                valorFila(
                  fila,
                  idxEstadoComprobante
                ) || ""
              ).trim().toUpperCase()
            : "";
        const estadoRendicion =
          idxEstadoRendicion >= 0
            ? String(
                valorFila(
                  fila,
                  idxEstadoRendicion
                ) || "PENDIENTE"
              ).trim().toUpperCase()
            : "PENDIENTE";

        if (
          estadoComprobante === "AUTORIZADO" ||
          (
            !estadoComprobante &&
            estadoRendicion === "AUTORIZADA"
          )
        ) {
          item.monto_rendido += monto;
        } else if (
          estadoComprobante === "RECHAZADO" ||
          estadoRendicion === "RECHAZADA"
        ) {
          item.monto_rechazado += monto;
        } else {
          item.monto_pendiente_revision += monto;
        }
      });
    }
  }

  Object.keys(resultado).forEach((rut) => {
    resultado[rut].saldo =
      resultado[rut].monto_asignado -
      resultado[rut].monto_rendido;
  });

  return resultado;
}


// Compatibilidad temporal: se conserva esta función antigua,
// pero el endpoint de colaboradores ya no la utiliza.
function obtenerFinancierosColaboradores(
  hojaConsolidado
) {
  const resultado = {};

  if (
    !hojaConsolidado ||
    hojaConsolidado.getLastRow() < 2
  ) {
    return resultado;
  }

  const datos =
    hojaConsolidado
      .getDataRange()
      .getDisplayValues();

  const encabezados = datos[0];

  const idxColaborador = buscarIndice(
    encabezados,
    "COLABORADOR"
  );

  const idxMontoAsignado = buscarIndice(
    encabezados,
    "MONTO ASIGNADO ACUMULADO"
  );

  const idxSaldo = buscarIndice(
    encabezados,
    "SALDO"
  );

  const idxSumaRendiciones = buscarIndice(
    encabezados,
    "SUMA RENDICIONES"
  );

  if (
    idxColaborador < 0 ||
    idxMontoAsignado < 0 ||
    idxSaldo < 0 ||
    idxSumaRendiciones < 0
  ) {
    return resultado;
  }

  for (
    let i = 1;
    i < datos.length;
    i++
  ) {
    const fila = datos[i];

    const colaborador = String(
      valorFila(
        fila,
        idxColaborador
      ) || ""
    ).trim();

    const rut =
      normalizarRutPermisos(
        extraerRut(colaborador)
      );

    if (!rut) {
      continue;
    }

    resultado[rut] = {
      monto_asignado:
        convertirNumero(
          valorFila(
            fila,
            idxMontoAsignado
          )
        ),
      monto_rendido:
        convertirNumero(
          valorFila(
            fila,
            idxSumaRendiciones
          )
        ),
      saldo:
        convertirNumero(
          valorFila(
            fila,
            idxSaldo
          )
        )
    };
  }

  return resultado;
}


function obtenerResumenRendicionesPorRut(
  hojaRespuestas
) {
  const resultado = {};

  if (
    !hojaRespuestas ||
    hojaRespuestas.getLastRow() < 2
  ) {
    return resultado;
  }

  const datos = obtenerDatosHojaConsultaCached(
    hojaRespuestas,
    "RESPUESTAS"
  );

  const encabezados = datos[0];

  const idxId = buscarIndiceCualquiera(
    encabezados,
    ["ID_Rendicion", "ID"]
  );

  const idxColaborador = buscarIndice(
    encabezados,
    "colaborador"
  );

  const idxRut = buscarIndice(
    encabezados,
    "rut"
  );

  const idxProcesadoEn = buscarIndice(
    encabezados,
    "procesado_en"
  );

  if (idxId < 0) {
    return resultado;
  }

  for (
    let i = 1;
    i < datos.length;
    i++
  ) {
    const fila = datos[i];

    const id = String(
      valorFila(fila, idxId) || ""
    ).trim();

    if (!id) {
      continue;
    }

    const colaborador =
      idxColaborador >= 0
        ? String(
            valorFila(
              fila,
              idxColaborador
            ) || ""
          ).trim()
        : "";

    let rut =
      extraerRut(colaborador);

    if (
      !rut &&
      idxRut >= 0
    ) {
      rut = valorFila(
        fila,
        idxRut
      );
    }

    const rutNormalizado =
      normalizarRutPermisos(rut);

    if (!rutNormalizado) {
      continue;
    }

    const procesadoEn =
      idxProcesadoEn >= 0
        ? String(
            valorFila(
              fila,
              idxProcesadoEn
            ) || ""
          ).trim()
        : "";

    if (!resultado[rutNormalizado]) {
      resultado[rutNormalizado] = {
        ids: {},
        cantidad_rendiciones: 0,
        ultima_rendicion_id: "",
        ultima_rendicion_en: "",
        ultima_rendicion_tiempo:
          Number.NEGATIVE_INFINITY
      };
    }

    const resumen =
      resultado[rutNormalizado];

    if (!resumen.ids[id]) {
      resumen.ids[id] = true;
      resumen.cantidad_rendiciones++;
    }

    const tiempo =
      convertirFechaComparableColaborador(
        procesadoEn
      );

    if (
      tiempo >
      resumen.ultima_rendicion_tiempo
    ) {
      resumen.ultima_rendicion_tiempo =
        tiempo;
      resumen.ultima_rendicion_id = id;
      resumen.ultima_rendicion_en =
        procesadoEn;
    }
  }

  Object.keys(resultado).forEach(
    (rut) => {
      delete resultado[rut].ids;
      delete resultado[rut]
        .ultima_rendicion_tiempo;
    }
  );

  return resultado;
}


function convertirFechaComparableColaborador(
  valor
) {
  if (!valor) {
    return Number.NEGATIVE_INFINITY;
  }

  const texto =
    String(valor).trim();

  const fechaChilena = texto.match(
    /^(\d{1,2})[\/-](\d{1,2})[\/-](\d{4})(?:[ T](\d{1,2}):(\d{2})(?::(\d{2}))?)?$/
  );

  if (fechaChilena) {
    return new Date(
      Number(fechaChilena[3]),
      Number(fechaChilena[2]) - 1,
      Number(fechaChilena[1]),
      Number(fechaChilena[4] || 0),
      Number(fechaChilena[5] || 0),
      Number(fechaChilena[6] || 0)
    ).getTime();
  }

  const tiempo = Date.parse(texto);

  return Number.isNaN(tiempo)
    ? Number.NEGATIVE_INFINITY
    : tiempo;
}

function obtenerListadoRendiciones(
  hoja,

  hojaComprobantesDuplicados
) {

    permisos,
    empresaSeleccionada
  );

  const cacheado = leerCacheJSON(cacheKey);
  if (cacheado && Array.isArray(cacheado)) {
    return responderJSON(cacheado);
  }

  const ultimaFila = hoja.getLastRow();
  const ultimaColumna = hoja.getLastColumn();

  if (ultimaFila < 2) {
    const gruposSinRespuestas = {};

    integrarRendicionesActualesDuplicadasEnListado(
      gruposSinRespuestas,
      hojaComprobantesDuplicados,
      permisos,
      empresaSeleccionada,
      hojaColaboradores
    );

    const resultadoVacio = Object.values(gruposSinRespuestas);
    guardarCacheJSON(cacheKey, resultadoVacio, 30);
    return responderJSON(resultadoVacio);
  }

  const encabezados = hoja
    .getRange(
      1,
      1,
      1,
      ultimaColumna
    )
    .getDisplayValues()[0];

  const idxId = buscarIndiceCualquiera(
    encabezados,
    ["ID_Rendicion", "ID"]
  );

  const idxColaborador = buscarIndice(
    encabezados,
    "colaborador"
  );

  const idxRut = buscarIndice(
    encabezados,
    "rut"
  );

  const idxViaje = buscarIndice(
    encabezados,
    "numero_viaje"
  );

  const idxMonto = buscarIndiceCualquiera(
    encabezados,
    ["monto_total", "MONTO"]
  );

  const idxProcesadoEn = buscarIndice(
    encabezados,
    "procesado_en"
  );

  // Respaldo para instalaciones donde procesado_en quedó vacío.
  // buscarIndiceCualquiera usa normalización de encabezados, por lo que
  // "Submission Date" y "submission_date" se consideran equivalentes.
  const idxFechaAlternativa = buscarIndiceCualquiera(
    encabezados,
    [
      "Submission Date",
      "submission_date",
      "submitted_at",
      "submittedAt",
      "Last Update Date",
      "last_update_date"
    ]
  );

  const idxEstadoRendicion = buscarIndice(
    encabezados,
    "estado_rendicion"
  );

  const idxAutorizadoPor = buscarIndice(
    encabezados,
    "autorizado_por"
  );

  const idxAutorizadoNombre = buscarIndice(
    encabezados,
    "autorizado_nombre"
  );

  const idxAutorizadoEn = buscarIndice(
    encabezados,
    "autorizado_en"
  );

  if (idxId === -1) {
    return responderJSON({
      error:
        'No se encontró la columna "ID_Rendicion" ni "ID" en RESPUESTAS.'
    });
  }

  const datos = hoja
    .getRange(
      2,
      1,
      ultimaFila - 1,
      ultimaColumna
    )
    .getDisplayValues();

  const tienePermisos =
    permisos &&
    permisos.encontrado;

  const esSuperAdmin =
    tienePermisos &&
    esRolSuperAdmin(permisos);

  const rutsPermitidos =
    tienePermisos &&
    Array.isArray(permisos.colaboradores)
      ? permisos.colaboradores.map(
          normalizarRutPermisos
        )
      : [];

  const grupos = {};

  const empresasPorRut =
    obtenerEmpresasPorRut(
      hojaColaboradores
    );

  datos.forEach((fila) => {
    const id = String(
      valorFila(
        fila,
        idxId
      ) || ""
    ).trim();

    if (!id) {
      return;
    }

    const colaborador =
      idxColaborador >= 0
        ? valorFila(
            fila,
            idxColaborador
          )
        : "";

    let rut =
  colaborador
    ? extraerRut(colaborador)
    : "";

if (!rut && idxRut >= 0) {
  rut = valorFila(
    fila,
    idxRut
  );
}

    const rutNormalizado =
      normalizarRutPermisos(
        rut
      );

    if (
      tienePermisos &&
      !esSuperAdmin
    ) {
      if (
        !rutNormalizado ||
        !rutsPermitidos.includes(
          rutNormalizado
        )
      ) {
        return;
      }
    }

    const empresaRendicion =
      empresasPorRut[rutNormalizado] || "";

    if (
      !empresaPermitidaParaConsulta(
        empresaRendicion,
        permisos,
        empresaSeleccionada
      )
    ) {
      return;
    }

    const numeroViaje =
      idxViaje >= 0
        ? valorFila(
            fila,
            idxViaje
          )
        : "";

    const montoTotal =
      idxMonto >= 0
        ? convertirNumero(valorFila(fila, idxMonto))
        : 0;

    const procesadoEn =
      obtenerFechaRendicionDesdeFila(
        fila,
        idxProcesadoEn,
        idxFechaAlternativa
      );

    const estadoRendicion =
      idxEstadoRendicion >= 0
        ? String(
            valorFila(
              fila,
              idxEstadoRendicion
            ) || "PENDIENTE"
          ).trim().toUpperCase()
        : "PENDIENTE";

    const autorizadoPor =
      idxAutorizadoPor >= 0
        ? valorFila(
            fila,
            idxAutorizadoPor
          )
        : "";

    const autorizadoNombre =
      idxAutorizadoNombre >= 0
        ? valorFila(
            fila,
            idxAutorizadoNombre
          )
        : "";

    const autorizadoEn =
      idxAutorizadoEn >= 0
        ? valorFila(
            fila,
            idxAutorizadoEn
          )
        : "";

    if (!grupos[id]) {
      grupos[id] = {
        ID: id,
        procesado_en: procesadoEn,
        colaborador: colaborador,
        rut: rut,
        empresa: empresaRendicion,
        numero_viaje: numeroViaje,
        monto_total: 0,
        estado_rendicion: estadoRendicion,
        autorizado_por: autorizadoPor,
        autorizado_nombre: autorizadoNombre,
        autorizado_en: autorizadoEn,
        cantidad_documentos: 0
      };
    }

    // Algunas rendiciones tienen la primera fila del grupo
    // sin procesado_en. Conservamos exclusivamente este campo
    // y tomamos el primer valor no vacío del mismo folio.
    if (
      !grupos[id].procesado_en &&
      procesadoEn
    ) {
      grupos[id].procesado_en =
        procesadoEn;
    }

    grupos[id].cantidad_documentos++;
    grupos[id].monto_total =
      Number(grupos[id].monto_total || 0) + montoTotal;
  });

  integrarRendicionesActualesDuplicadasEnListado(
    grupos,
    hojaComprobantesDuplicados,
    permisos,
    empresaSeleccionada,
    hojaColaboradores
  );

  const resultado = Object.values(grupos);

  guardarCacheJSON(cacheKey, resultado, 30);

  return responderJSON(resultado);
}



// ======================================================
// FECHA DE RENDICIÓN
// ======================================================

function obtenerFechaRendicionDesdeFila(
  fila,
  idxProcesadoEn,
  idxFechaAlternativa
) {
  let valor = "";

  if (idxProcesadoEn >= 0) {
    valor = String(
      valorFila(
        fila,
        idxProcesadoEn
      ) || ""
    ).trim();
  }

  // Si procesado_en está vacío, usar fecha nativa/original
  // de Jotform como respaldo.
  if (
    !valor &&
    idxFechaAlternativa >= 0
  ) {
    valor = String(
      valorFila(
        fila,
        idxFechaAlternativa
      ) || ""
    ).trim();
  }

  return valor;
}

      ) || "",

    autorizado_nombre:
      valorFila(
        primeraFilaCoincidente || [],
        idx.autorizado_nombre
      ) || "",

    autorizado_en:
      valorFila(
        primeraFilaCoincidente || [],
        idx.autorizado_en
      ) || "",

    cantidad_documentos:
      documentos.length,

    total_presentado:
      totalesRevision.presentado,

    total_autorizado:
      totalesRevision.autorizado,

    total_rechazado:
      totalesRevision.rechazado,

    total_pendiente_revision:
      totalesRevision.pendiente,

    cantidad_duplicados:
      auditoriaDuplicados.cantidad,

    total_duplicados:
      auditoriaDuplicados.total,

    documentos_duplicados:
      auditoriaDuplicados.documentos,

    documentos:
      documentos
  });
}

function obtenerIdsRendicionesEliminadas(libro) {
  const eliminadas = {};
  if (!libro) return eliminadas;

  const hoja = libro.getSheetByName("AUDITORIA_RENDICIONES");
  if (!hoja || hoja.getLastRow() < 2) return eliminadas;

  const ultimaFila = hoja.getLastRow();
  const ultimaColumna = Math.min(hoja.getLastColumn(), 3);
  const datos = hoja.getRange(2, 1, ultimaFila - 1, ultimaColumna).getDisplayValues();

  datos.forEach((fila) => {
    const accion = String(fila[1] || "").trim().toUpperCase();
    const id = String(fila[2] || "").trim();
    if (accion === "ELIMINACION_RENDICION" && id) {
      eliminadas[id] = true;
    }
  });

  return eliminadas;
}


function integrarRendicionesActualesDuplicadasEnListado(
  grupos,
  hojaComprobantesDuplicados,
  permisos,
  empresaSeleccionada,
  hojaColaboradores
) {
  if (
    !hojaComprobantesDuplicados ||
    hojaComprobantesDuplicados.getLastRow() < 2
  ) {
    return;
  }

  const ultimaFila =
    hojaComprobantesDuplicados.getLastRow();
  const ultimaColumna =
    hojaComprobantesDuplicados.getLastColumn();
  const valores = hojaComprobantesDuplicados
    .getRange(1, 1, ultimaFila, ultimaColumna)
    .getDisplayValues();
  const encabezados = valores[0];
  const idx = {
    id_actual: buscarIndice(encabezados, "id_rendicion_actual"),
    detectado_en: buscarIndice(encabezados, "detectado_en"),
    colaborador: buscarIndice(encabezados, "colaborador_actual"),
    rut: buscarIndice(encabezados, "rut_actual"),
    empresa: buscarIndice(encabezados, "empresa_actual"),
    monto: buscarIndice(encabezados, "monto_total")
  };

  if (idx.id_actual < 0) {
    return;
  }

  const tienePermisos =
    permisos && permisos.encontrado;
  const esSuperAdmin =
    tienePermisos && esRolSuperAdmin(permisos);
  const rutsPermitidos =
    tienePermisos && Array.isArray(permisos.colaboradores)
      ? permisos.colaboradores.map(normalizarRutPermisos)
      : [];
  const empresasPorRut = obtenerEmpresasPorRut(hojaColaboradores);
  const idsEliminadas = obtenerIdsRendicionesEliminadas(
    hojaComprobantesDuplicados.getParent()
  );

  for (let i = 1; i < valores.length; i++) {
    const fila = valores[i];
    const idActual = String(
      valorFila(fila, idx.id_actual) || ""
    ).trim();

    if (!idActual || idsEliminadas[idActual]) {
      continue;
    }

    const colaborador = String(
      valorFila(fila, idx.colaborador) || ""
    ).trim();
    const rutAuditoria = String(
      valorFila(fila, idx.rut) || ""
    ).trim();
    const rut = String(
      extraerRut(rutAuditoria) ||
      extraerRut(colaborador) ||
      rutAuditoria ||
      ""
    ).trim();
    const rutNormalizado =
      normalizarRutPermisos(rut);
    const empresa = String(
      valorFila(fila, idx.empresa) ||
      empresasPorRut[rutNormalizado] ||
      ""
    ).trim();

    if (
      tienePermisos &&
      !esSuperAdmin &&
      (
        !rutNormalizado ||
        !rutsPermitidos.includes(rutNormalizado)
      )
    ) {
      continue;
    }

    if (
      !empresaPermitidaParaConsulta(
        empresa,
        permisos,
        empresaSeleccionada
      )
    ) {
      continue;
    }

    if (!grupos[idActual]) {
      grupos[idActual] = {
        ID: idActual,
        procesado_en:
          valorFila(fila, idx.detectado_en),
        colaborador: colaborador,
        rut: rut,
        empresa: empresa,
        numero_viaje: "",
        monto_total: 0,
        estado_rendicion: "EXCLUIDA_POR_DUPLICADOS",
        estado_operativo: "SOLO_DUPLICADOS",
        solo_duplicados: true,
        autorizado_por: "",
        autorizado_nombre: "",
        autorizado_en: "",
        cantidad_documentos: 0,
        cantidad_duplicados: 0,
        total_duplicados: 0
      };
    }

    grupos[idActual].cantidad_duplicados =
      Number(grupos[idActual].cantidad_duplicados || 0) + 1;
    grupos[idActual].total_duplicados =
      Number(grupos[idActual].total_duplicados || 0) +
      convertirNumero(valorFila(fila, idx.monto));
    grupos[idActual].monto_total = grupos[idActual].total_duplicados;
  }
}

  hojaColaboradores,

  permisos,
  empresaSeleccionada
) {
  const auditoria = obtenerDuplicadosDeRendicion(
    hojaComprobantesDuplicados,
    idBuscado,
    ""
  );

  if (!auditoria.cantidad) {
    return responderJSON({
      error: "No se encontró la rendición solicitada.",
      ID: idBuscado
    });
  }

  const colaborador =
    auditoria.colaborador_actual || "";
  const rutAuditoria =
    auditoria.rut_actual || "";
  const rut =
    extraerRut(rutAuditoria) ||
    extraerRut(colaborador) ||
    rutAuditoria ||
    "";

  if (permisos && permisos.encontrado && !esRolSuperAdmin(permisos)) {
    const rutNormalizado = normalizarRutPermisos(rut);
    const rutsPermitidos = Array.isArray(permisos.colaboradores)
      ? permisos.colaboradores.map(normalizarRutPermisos)
      : [];

    if (
      !rutNormalizado ||
      !rutsPermitidos.includes(rutNormalizado)
    ) {
      return responderJSON({
        error: "No autorizado para ver esta rendición.",
        ID: idBuscado
      });
    }
  }

    obtenerDatosColaborador(hojaColaboradores, rut);
  const empresa =
    auditoria.empresa_actual ||
    datosColaborador.empresa ||
    "";

  if (
    !empresaPermitidaParaConsulta(
      empresa,
      permisos,
      empresaSeleccionada
    )
  ) {
    return responderJSON({
      error: "No autorizado para consultar la empresa de esta rendición.",
      ID: idBuscado
    });
  }

  const datosFinancieros = obtenerDatosConsolidado(
    hojaConsolidado,
    rut,
    colaborador
  );

  return responderJSON({

    procesado_en: auditoria.detectado_en || "",

    colaborador: colaborador,
    rut: rut,

    centro_costo: datosColaborador.centro_costo || "",
    monto_asignado: datosFinancieros.monto_asignado,
    saldo_2: datosFinancieros.saldo_2,
    estado_rendicion: "EXCLUIDA_POR_DUPLICADOS",
    estado_operativo: "SOLO_DUPLICADOS",
    solo_duplicados: true,
    puede_revisar: false,
    autorizado_por: "",
    autorizado_nombre: "",
    autorizado_en: "",
    cantidad_documentos: 0,
    total_presentado: 0,
    total_autorizado: 0,
    total_rechazado: 0,
    total_pendiente_revision: 0,
    cantidad_duplicados: auditoria.cantidad,
    total_duplicados: auditoria.total,
    documentos_duplicados: auditoria.documentos,
    documentos: []
  });
}

) {

    total: 0,
    documentos: [],
    id_rendicion_actual: "",
    submission_id_actual: "",
    colaborador_actual: "",
    rut_actual: "",
    empresa_actual: "",
    detectado_en: ""
  };

  if (
    !hojaComprobantesDuplicados
  ) {
    return resultado;
  }

  const ultimaFila =
    hojaComprobantesDuplicados.getLastRow();

  const ultimaColumna =
    hojaComprobantesDuplicados.getLastColumn();

  if (
    ultimaFila < 2 ||
    ultimaColumna < 1
  ) {
    return resultado;
  }

  const encabezados =
    hojaComprobantesDuplicados
      .getRange(1, 1, 1, ultimaColumna)
      .getDisplayValues()[0];

  const indices = {
    id_duplicado:
      buscarIndice(
        encabezados,
        "id_duplicado"
      ),

    detectado_en:
      buscarIndice(
        encabezados,
        "detectado_en"
      ),

    submission_id_actual:
      buscarIndice(
        encabezados,
        "submission_id_actual"
      ),

    colaborador_actual:
      buscarIndice(
        encabezados,
        "colaborador_actual"
      ),

    rut_actual:
      buscarIndice(
        encabezados,
        "rut_actual"
      ),

    empresa_actual:
      buscarIndice(
        encabezados,
        "empresa_actual"
      ),

    numero_documento:
      buscarIndice(
        encabezados,
        "numero_documento"
      ),

    fecha_documento:
      buscarIndice(
        encabezados,
        "fecha_documento"
      ),

    hora_documento:
      buscarIndice(
        encabezados,
        "hora_documento"
      ),

    monto_total:
      buscarIndice(
        encabezados,
        "monto_total"
      ),

    proveedor:
      buscarIndice(
        encabezados,
        "proveedor"
      ),

    archivo_url_actual:
      buscarIndice(
        encabezados,
        "archivo_url_actual"
      ),

    id_rendicion_original:
      buscarIndice(
        encabezados,
        "id_rendicion_original"
      ),

    colaborador_original:
      buscarIndice(
        encabezados,
        "colaborador_original"
      ),

    fotografia_original:
      buscarIndice(
        encabezados,
        "fotografia_original"
      ),

    clave_duplicado:
      buscarIndice(
        encabezados,
        "clave_duplicado"
      ),

    tipo_duplicado:
      buscarIndice(
        encabezados,
        "tipo_duplicado"
      ),

    estado:
      buscarIndice(
        encabezados,
        "estado"
      ),

    id_rendicion_actual:
      buscarIndice(
        encabezados,
        "id_rendicion_actual"
      )
  };

  if (
    indices.id_rendicion_actual < 0 &&
    indices.submission_id_actual < 0
  ) {
    return resultado;
  }

  const submissionBuscado = String(
    submissionIdActual || ""
  ).trim();
  const idActualBuscado = String(
    idRendicionActual || ""
  ).trim();

  let filasCoincidentes = [];

  if (idActualBuscado && indices.id_rendicion_actual >= 0) {
    filasCoincidentes = buscarFilasPorValorExacto(
      hojaComprobantesDuplicados,
      indices.id_rendicion_actual,
      idActualBuscado
    );
  }

  if (
    !filasCoincidentes.length &&
    submissionBuscado &&
    indices.submission_id_actual >= 0

    filasCoincidentes = buscarFilasPorValorExacto(
      hojaComprobantesDuplicados,

      Boolean(idActualBuscado) &&
      idActualFila === idActualBuscado;

    const coincideSubmissionAntiguo =
      !idActualFila &&
      Boolean(submissionBuscado) &&
      submissionFila === submissionBuscado;

    if (
      !coincideIdActual &&
      !coincideSubmissionAntiguo
    ) {
      continue;
    }

    const monto = convertirNumero(
      valorFila(
        fila,
        indices.monto_total
      )
    );

      resultado.id_rendicion_actual = idActualFila;
      resultado.submission_id_actual = submissionFila;
      resultado.colaborador_actual = valorFila(
        fila,
        indices.colaborador_actual

      resultado.rut_actual = valorFila(
        fila,
        indices.rut_actual
      );
      resultado.empresa_actual = valorFila(
        fila,
        indices.empresa_actual
      );
      resultado.detectado_en = valorFila(
        fila,
        indices.detectado_en
      );
    }

    resultado.documentos.push({
      id_duplicado:
        valorFila(
          fila,
          indices.id_duplicado
        ),

      detectado_en:
        valorFila(
          fila,
          indices.detectado_en
        ),

      submission_id_actual:
        submissionFila,

      id_rendicion_actual:
        idActualFila,

      colaborador_actual:
        valorFila(
          fila,
          indices.colaborador_actual
        ),

      rut_actual:
        valorFila(
          fila,
          indices.rut_actual
        ),

      empresa_actual:
        valorFila(
          fila,
          indices.empresa_actual
        ),

      numero_documento:
        valorFila(
          fila,
          indices.numero_documento
        ),

      fecha_documento:
        valorFila(
          fila,
          indices.fecha_documento
        ),

      hora_documento:
        valorFila(
          fila,
          indices.hora_documento
        ),

      monto_total:
        monto,

      proveedor:
        valorFila(
          fila,
          indices.proveedor
        ),

      archivo_url_actual:
        valorFila(
          fila,
          indices.archivo_url_actual
        ),

      id_rendicion_original:
        valorFila(
          fila,
          indices.id_rendicion_original
        ),

      colaborador_original:
        valorFila(
          fila,
          indices.colaborador_original
        ),

      fotografia_original:
        valorFila(
          fila,
          indices.fotografia_original
        ),

      clave_duplicado:
        valorFila(
          fila,
          indices.clave_duplicado
        ),

      tipo_duplicado:
        valorFila(
          fila,
          indices.tipo_duplicado
        ),

      estado:
        valorFila(
          fila,
          indices.estado
        )
    });

    resultado.total += monto;
  }

  resultado.cantidad =
    resultado.documentos.length;

  return resultado;
}

function obtenerMonitorSistema(
  libro,
  hojaRespuestas,
  hojaComprobantesDuplicados,
  permisos
) {
  if (!permisos || !permisos.encontrado || !permisos.activo) {
    return responderJSON({ ok: false, error: "Usuario no autorizado." });
  }

  if (!esRolSuperAdmin(permisos)) {
    return responderJSON({
      ok: false,
      error: "Solo el Super Administrador puede consultar el monitor del sistema."
    });
  }

  const cacheKey = obtenerClaveCacheConsulta("MONITOR_SISTEMA", permisos, "");
  const cacheado = leerCacheJSON(cacheKey);
  if (cacheado && cacheado.ok) {
    return responderJSON(cacheado);
  }

  const resultado = {
    ok: true,
    generado_en: new Date().toISOString(),
    rendiciones: { total: 0, pendientes: 0, autorizadas: 0, rechazadas: 0 },
    comprobantes: { total: 0, duplicados_historicos: 0 },
    ultima_actividad: "",
    auditoria_reciente: []
  };

  if (hojaRespuestas && hojaRespuestas.getLastRow() >= 2) {
    const datos = obtenerDatosHojaConsultaCached(hojaRespuestas, "RESPUESTAS");
    const encabezados = datos[0] || [];
    const idxId = buscarIndiceCualquiera(encabezados, ["ID_Rendicion", "ID"]);
    const idxEstado = buscarIndice(encabezados, "estado_rendicion");
    const idxProcesado = buscarIndice(encabezados, "procesado_en");
    const idxAlternativa = buscarIndiceCualquiera(encabezados, [
      "Submission Date", "submission_date", "submitted_at", "submittedAt",
      "Last Update Date", "last_update_date"
    ]);

    const porId = {};
    let ultimaMs = Number.NEGATIVE_INFINITY;

    datos.slice(1).forEach((fila) => {
      const id = idxId >= 0 ? String(valorFila(fila, idxId) || "").trim() : "";
      if (!id) return;

      resultado.comprobantes.total++;
      if (!porId[id]) {
        const estado = idxEstado >= 0
          ? String(valorFila(fila, idxEstado) || "PENDIENTE").trim().toUpperCase()
          : "PENDIENTE";
        porId[id] = estado;
      }

      const fecha = obtenerFechaRendicionDesdeFila(fila, idxProcesado, idxAlternativa);
      const ms = convertirFechaComparableColaborador(fecha);
      if (ms > ultimaMs) {
        ultimaMs = ms;
        resultado.ultima_actividad = fecha || "";
      }
    });

    Object.keys(porId).forEach((id) => {
      resultado.rendiciones.total++;
      const estado = porId[id];
      if (estado === "AUTORIZADA") resultado.rendiciones.autorizadas++;
      else if (estado === "RECHAZADA") resultado.rendiciones.rechazadas++;
      else resultado.rendiciones.pendientes++;
    });
  }

  if (hojaComprobantesDuplicados && hojaComprobantesDuplicados.getLastRow() >= 2) {
    resultado.comprobantes.duplicados_historicos =
      hojaComprobantesDuplicados.getLastRow() - 1;
  }

  const hojaAuditoria = libro.getSheetByName("AUDITORIA_RENDICIONES");
  if (hojaAuditoria && hojaAuditoria.getLastRow() >= 2) {
    const ultimaFila = hojaAuditoria.getLastRow();
    const cantidad = Math.min(20, ultimaFila - 1);
    const filas = hojaAuditoria
      .getRange(ultimaFila - cantidad + 1, 1, cantidad, 7)
      .getDisplayValues();

    resultado.auditoria_reciente = filas.reverse().map((fila) => ({
      fecha: fila[0] || "",
      accion: fila[1] || "",
      id_rendicion: fila[2] || "",
      usuario: fila[3] || "",
      nombre_usuario: fila[4] || "",
      rol: fila[5] || "",
      detalle: fila[6] || ""
    }));
  }

  guardarCacheJSON(cacheKey, resultado, 15);
  return responderJSON(resultado);
}

function asegurarHojaAuditoriaRendiciones(libro) {
  let hoja = libro.getSheetByName("AUDITORIA_RENDICIONES");
  if (!hoja) {
    hoja = libro.insertSheet("AUDITORIA_RENDICIONES");
    hoja.appendRow(["FECHA", "ACCION", "ID_RENDICION", "USUARIO", "NOMBRE_USUARIO", "ROL", "DETALLE"]);
  }
  return hoja;
}

function registrarAuditoriaRendicion(libro, accion, id, permisos, detalle) {
  const hoja = asegurarHojaAuditoriaRendiciones(libro);
  hoja.appendRow([
    new Date(),
    accion,
    id,
    permisos && permisos.email ? permisos.email : "",
    permisos && permisos.nombre ? permisos.nombre : "",
    permisos && permisos.rol ? permisos.rol : "",
    detalle || ""
  ]);
}

function eliminarRendicionDesdeGet(hojaRespuestas, parametros, permisos) {
  const id = String(parametros.id || "").trim();
  if (!id) return responderJSON({ ok: false, error: "Debe indicar el ID de la rendición." });
  if (!permisos || !permisos.encontrado || !permisos.activo) return responderJSON({ ok: false, error: "Usuario no autorizado." });
  if (!esRolSuperAdmin(permisos)) return responderJSON({ ok: false, error: "Solo el Super Administrador puede eliminar rendiciones." });

  const bloqueo = LockService.getScriptLock();

  try {
    bloqueo.waitLock(30000);

    const libro = hojaRespuestas.getParent();
    const hojaDuplicados = libro.getSheetByName("COMPROBANTES_DUPLICADOS");
    const ultimaFila = hojaRespuestas.getLastRow();
    const ultimaColumna = hojaRespuestas.getLastColumn();

    let filas = [];
    if (ultimaFila >= 2 && ultimaColumna >= 1) {
      const encabezados = hojaRespuestas.getRange(1, 1, 1, ultimaColumna).getDisplayValues()[0];
      const idxId = buscarIndiceCualquiera(encabezados, ["ID_Rendicion", "ID"]);
      if (idxId < 0) return responderJSON({ ok: false, error: "No existe la columna de ID de rendición." });

      filas = buscarFilasPorValorExacto(
        hojaRespuestas,
        idxId,
        id
      );
    }

    const auditoriaDuplicados = obtenerDuplicadosDeRendicion(
      hojaDuplicados,
      id,
      ""
    );

    if (!filas.length && !auditoriaDuplicados.cantidad) {
      return responderJSON({ ok: false, error: "No se encontró la rendición." });
    }

    registrarAuditoriaRendicion(
      libro,
      "ELIMINACION_RENDICION",
      id,
      permisos,
      filas.length
        ? `Se eliminaron ${filas.length} fila(s) de RESPUESTAS. El historial de duplicados se conserva solo como auditoría y ya no se mostrará como rendición activa.`
        : "La rendición ya no tenía filas en RESPUESTAS. Se marcó como eliminada para que el historial de duplicados no la vuelva a mostrar en el sistema."
    );

    filas.sort((a, b) => b - a).forEach((numeroFila) => hojaRespuestas.deleteRow(numeroFila));
    SpreadsheetApp.flush();
    invalidarCacheListadoRendiciones();

    return responderJSON({
      ok: true,
      success: true,
      id: id,
      filas_eliminadas: filas.length,
      historial_duplicados_conservado: true,
      oculto_del_listado: true
    });
  } catch (error) {
    return responderJSON({
      ok: false,
      error: "No fue posible eliminar la rendición: " + error.message
    });
  } finally {
    try {
      bloqueo.releaseLock();
    } catch (errorBloqueo) {
      // El bloqueo pudo expirar antes de liberarse.
    }
  }
}


// ======================================================
// AUTORIZACIÓN DE RENDICIONES
// ======================================================

function procesarAutorizacionDesdeGet(
  hojaRespuestas,
  parametros,
  permisos
) {
  const id = String(
    parametros.id || ""
  ).trim();

  if (!id) {
    return responderJSON({
      ok: false,
      error:
        "Debe indicar el ID de la rendición."
    });
  }

  if (
    !permisos ||
    !permisos.encontrado ||
    !permisos.activo
  ) {
    return responderJSON({
      ok: false,
      error:
        "Usuario no autorizado."
    });
  }

  if (!esRolAutorizador(permisos)) {
    return responderJSON({
      ok: false,
      error:
        "Solo el Jefe de Área puede autorizar rendiciones."
    });
  }

  asegurarEstructuraRevisionComprobantes(
    hojaRespuestas
  );

  asegurarIdsComprobantesRendicion(
    hojaRespuestas,
    id
  );

  const ultimaFila =
    hojaRespuestas.getLastRow();

  const ultimaColumna =
    hojaRespuestas.getLastColumn();

  if (ultimaFila < 2) {
    return responderJSON({
      ok: false,
      error:
        "No existen rendiciones."
    });
  }

  const encabezados =
    hojaRespuestas
      .getRange(
        1,
        1,
        1,
        ultimaColumna
      )
      .getDisplayValues()[0];

  const idxId =
    buscarIndiceCualquiera(
      encabezados,
      ["ID_Rendicion", "ID"]
    );

  const idxEstado =
    buscarIndice(
      encabezados,
      "estado_rendicion"
    );

  const idxAutorizadoPor =
    buscarIndice(
      encabezados,
      "autorizado_por"
    );

  const idxAutorizadoNombre =
    buscarIndice(
      encabezados,
      "autorizado_nombre"
    );

  const idxAutorizadoEn =
    buscarIndice(
      encabezados,
      "autorizado_en"
    );

  const idxColaborador =
    buscarIndice(
      encabezados,
      "colaborador"
    );

  const idxRut =
    buscarIndice(
      encabezados,
      "rut"
    );

  const idxEstadoComprobante =
    buscarIndice(
      encabezados,
      "estado_comprobante"
    );

  const idxRevisionMotivo =
    buscarIndice(
      encabezados,
      "revision_motivo"
    );

  const idxRevisadoPor =
    buscarIndice(
      encabezados,
      "revisado_por"
    );

  const idxRevisadoNombre =
    buscarIndice(
      encabezados,
      "revisado_nombre"
    );

  const idxRevisadoEn =
    buscarIndice(
      encabezados,
      "revisado_en"
    );

  if (idxId < 0) {
    return responderJSON({
      ok: false,
      error:
        "No existe la columna de ID de rendición."
    });
  }

  const filas = buscarFilasPorValorExacto(
    hojaRespuestas,
    idxId,
    id
  );

  if (!filas.length) {
    return responderJSON({
      ok: false,
      error:
        "No se encontró la rendición.",
      ID: id
    });
  }

  const filaRendicion = hojaRespuestas
    .getRange(filas[0], 1, 1, ultimaColumna)
    .getDisplayValues()[0];

  const colaboradorRendicion =
    idxColaborador >= 0
      ? valorFila(
          filaRendicion,
          idxColaborador
        )
      : "";

  const rutRendicion =
    normalizarRutPermisos(
      extraerRut(
        colaboradorRendicion
      ) ||
      (
        idxRut >= 0
          ? valorFila(
              filaRendicion,
              idxRut
            )
          : ""
      )
    );

  if (
    !esRolSuperAdmin(permisos) &&
    !usuarioTieneRutAsignado(
      permisos,
      rutRendicion
    )
  ) {
    return responderJSON({
      ok: false,
      error:
        "No puede autorizar una rendición de un colaborador no asignado.",
      ID: id
    });
  }

  const estadoActual =
    String(
      hojaRespuestas
        .getRange(
          filas[0],
          idxEstado + 1
        )
        .getDisplayValue() ||
      "PENDIENTE"
    )
      .trim()
      .toUpperCase();

  if (estadoActual === "AUTORIZADA") {
    return responderJSON({
      ok: true,
      ya_autorizada: true,
      ID: id,
      estado_rendicion:
        "AUTORIZADA",
      autorizado_por:
        hojaRespuestas
          .getRange(
            filas[0],
            idxAutorizadoPor + 1
          )
          .getDisplayValue(),
      autorizado_nombre:
        hojaRespuestas
          .getRange(
            filas[0],
            idxAutorizadoNombre + 1
          )
          .getDisplayValue(),
      autorizado_en:
        hojaRespuestas
          .getRange(
            filas[0],
            idxAutorizadoEn + 1
          )
          .getDisplayValue()
    });
  }

  const ahora = new Date();

  filas.forEach(
    (numeroFila) => {
      hojaRespuestas
        .getRange(
          numeroFila,
          idxEstado + 1
        )
        .setValue(
          "AUTORIZADA"
        );

      hojaRespuestas
        .getRange(
          numeroFila,
          idxAutorizadoPor + 1
        )
        .setValue(
          permisos.email || ""
        );

      hojaRespuestas
        .getRange(
          numeroFila,
          idxAutorizadoNombre + 1
        )
        .setValue(
          permisos.nombre || ""
        );

      hojaRespuestas
        .getRange(
          numeroFila,
          idxAutorizadoEn + 1
        )
        .setValue(
          ahora
        );

      hojaRespuestas
        .getRange(
          numeroFila,
          idxEstadoComprobante + 1
        )
        .setValue("AUTORIZADO");

      hojaRespuestas
        .getRange(
          numeroFila,
          idxRevisionMotivo + 1
        )
        .setValue("");

      hojaRespuestas
        .getRange(
          numeroFila,
          idxRevisadoPor + 1
        )
        .setValue(permisos.email || "");

      hojaRespuestas
        .getRange(
          numeroFila,
          idxRevisadoNombre + 1
        )
        .setValue(permisos.nombre || "");

      hojaRespuestas
        .getRange(
          numeroFila,
          idxRevisadoEn + 1
        )
        .setValue(ahora);
    }
  );

  SpreadsheetApp.flush();
  invalidarCacheListadoRendiciones();

  return responderJSON({
    ok: true,
    ID: id,
    estado_rendicion:
      "AUTORIZADA",
    autorizado_por:
      permisos.email || "",
    autorizado_nombre:
      permisos.nombre || "",
    autorizado_en:
      ahora.toISOString(),
    filas_actualizadas:
      filas.length
  });
}


function asegurarColumnasAutorizacion(
  hoja
) {
  const nombres = [
    "estado_rendicion",
    "autorizado_por",
    "autorizado_nombre",
    "autorizado_en"
  ];

  let encabezados =
    hoja
      .getRange(
        1,
        1,
        1,
        hoja.getLastColumn()
      )
      .getDisplayValues()[0];

  nombres.forEach(
    (nombre) => {
      if (
        buscarIndice(
          encabezados,
          nombre
        ) < 0
      ) {
        const nuevaColumna =
          hoja.getLastColumn() + 1;

        hoja
          .getRange(
            1,
            nuevaColumna
          )
          .setValue(
            nombre
          );

        encabezados =
          hoja
            .getRange(
              1,
              1,
              1,
              hoja.getLastColumn()
            )
            .getDisplayValues()[0];
      }
    }
  );
}


function obtenerValorPrimeraFilaRendicion(
  datos,
  idxId,
  idBuscado,
  idxCampo,
  valorDefecto
) {
  if (
    idxCampo === undefined ||
    idxCampo === null ||
    idxCampo < 0
  ) {
    return valorDefecto;
  }

  for (
    let i = 0;
    i < datos.length;
    i++
  ) {
    const idFila =
      String(
        valorFila(
          datos[i],
          idxId
        ) || ""
      ).trim();

    if (idFila === idBuscado) {
      const valor =
        valorFila(
          datos[i],
          idxCampo
        );

      return valor === ""
        ? valorDefecto
        : valor;
    }
  }

  return valorDefecto;
}


// ======================================================
// BÚSQUEDA DE COLUMNAS
// ======================================================

function buscarIndice(
  encabezados,
  nombreBuscado
) {
  const objetivo =
    normalizarEncabezado(
      nombreBuscado
    );

  return encabezados.findIndex(
    (encabezado) =>
      normalizarEncabezado(
        encabezado
      ) === objetivo
  );
}


function buscarIndiceCualquiera(
  encabezados,
  nombres
) {
  for (
    let i = 0;
    i < nombres.length;
    i++
  ) {
    const indice =
      buscarIndice(
        encabezados,
        nombres[i]
      );

    if (indice !== -1) {
      return indice;
    }
  }

  return -1;
}


function normalizarEncabezado(valor) {
  return String(valor || "")
    .trim()
    .toLowerCase()
    .replace(/\s+/g, "_");
}


function valorFila(
  fila,
  indice
) {
  if (indice < 0) {
    return "";
  }

  return fila[indice] ?? "";
}


// ======================================================
// RUT
// ======================================================

function extraerRut(texto) {
  if (!texto) {
    return "";
  }

  const partes =
    String(texto).split("/");

  return partes[0].trim();
}


function normalizarRutPermisos(rut) {
  return String(rut || "")
    .trim()
    .replace(/\./g, "")
    .replace(/-/g, "")
    .replace(/\s/g, "")
    .toUpperCase();
}


function normalizarRolSistema(rol) {
  return String(rol || "")
    .trim()
    .toUpperCase()
    .replace(/_/g, " ")
    .replace(/\s+/g, " ");
}


function esRolSuperAdmin(permisos) {
  return normalizarRolSistema(
    permisos && permisos.rol
  ) === "SUPER ADMIN";
}


function esRolAutorizador(permisos) {
  const rol = normalizarRolSistema(
    permisos && permisos.rol
  );

  return (
    rol === "ADMIN" ||
    rol === "SUPER ADMIN"
  );
}

function asegurarEstructuraRevisionComprobantes(
  hoja
) {
  asegurarColumnasAutorizacion(hoja);

  const nombres = [
    "id_comprobante",
    "estado_comprobante",
    "revision_motivo",
    "revisado_por",
    "revisado_nombre",
    "revisado_en"
  ];

  let encabezados = hoja
    .getRange(
      1,
      1,
      1,
      hoja.getLastColumn()
    )
    .getDisplayValues()[0];

  nombres.forEach((nombre) => {
    if (buscarIndice(encabezados, nombre) < 0) {
      hoja
        .getRange(
          1,
          hoja.getLastColumn() + 1
        )
        .setValue(nombre);

      encabezados = hoja
        .getRange(
          1,
          1,
          1,
          hoja.getLastColumn()
        )
        .getDisplayValues()[0];
    }
  });
}


function asegurarIdsComprobantesRendicion(
  hoja,
  idRendicion
) {
  const idBuscado = String(
    idRendicion || ""
  ).trim();

  if (!idBuscado || hoja.getLastRow() < 2) {
    return;
  }

  const bloqueo =
    LockService.getScriptLock();

  bloqueo.waitLock(30000);

  try {
    const ultimaColumna =
      hoja.getLastColumn();
    const encabezados = hoja
      .getRange(1, 1, 1, ultimaColumna)
      .getDisplayValues()[0];
    const idxId = buscarIndiceCualquiera(
      encabezados,
      ["ID_Rendicion", "ID"]
    );
    const idxIdComprobante = buscarIndice(
      encabezados,
      "id_comprobante"
    );
    const idxEstadoComprobante = buscarIndice(
      encabezados,
      "estado_comprobante"
    );
    const idxEstadoRendicion = buscarIndice(
      encabezados,
      "estado_rendicion"
    );

    if (
      idxId < 0 ||
      idxIdComprobante < 0 ||
      idxEstadoComprobante < 0
    ) {
      throw new Error(
        "No fue posible preparar la identificación de comprobantes."
      );
    }

    const filasCoincidentes = buscarFilasPorValorExacto(
      hoja,
      idxId,
      idBuscado
    );

    filasCoincidentes.forEach((numeroFila) => {
      const fila = hoja
        .getRange(numeroFila, 1, 1, ultimaColumna)
        .getDisplayValues()[0];

      const idActual = String(
        valorFila(fila, idxIdComprobante) || ""
      ).trim();

      if (!idActual) {
        hoja
          .getRange(numeroFila, idxIdComprobante + 1)
          .setValue(Utilities.getUuid());
      }

      const estadoActual = String(
        valorFila(fila, idxEstadoComprobante) || ""
      ).trim().toUpperCase();

      if (!estadoActual) {
        const estadoRendicion =
          idxEstadoRendicion >= 0
            ? String(
                valorFila(fila, idxEstadoRendicion) || "PENDIENTE"
              ).trim().toUpperCase()
            : "PENDIENTE";

        let estadoInicial = "PENDIENTE";
        if (estadoRendicion === "AUTORIZADA") {
          estadoInicial = "AUTORIZADO";
        } else if (estadoRendicion === "RECHAZADA") {
          estadoInicial = "RECHAZADO";
        }

        hoja
          .getRange(numeroFila, idxEstadoComprobante + 1)
          .setValue(estadoInicial);
      }
    });

    SpreadsheetApp.flush();

  } finally {
    bloqueo.releaseLock();
  }
}


function procesarRevisionComprobantesDesdeGet(
  hojaRespuestas,
  hojaColaboradores,
  parametros,
  permisos,
  empresaSeleccionada
) {
  const id = String(
    parametros.id || ""
  ).trim();

  if (!id) {
    return responderJSON({
      ok: false,
      error:
        "Debe indicar el ID de la rendición."
    });
  }

  if (!esRolAutorizador(permisos)) {
    return responderJSON({
      ok: false,
      error:
        "Solo el Jefe de Área o Super Administrador puede revisar comprobantes."
    });
  }

  let revisiones;

  try {
    revisiones = JSON.parse(
      String(
        parametros.revisiones || "[]"
      )
    );
  } catch (error) {
    return responderJSON({
      ok: false,
      error:
        "El contenido de las revisiones no es válido."
    });
  }

  if (
    !Array.isArray(revisiones) ||
    !revisiones.length
  ) {
    return responderJSON({
      ok: false,
      error:
        "Debe revisar todos los comprobantes antes de finalizar."
    });
  }

  asegurarEstructuraRevisionComprobantes(
    hojaRespuestas
  );
  asegurarIdsComprobantesRendicion(
    hojaRespuestas,
    id
  );

  const ultimaColumna =
    hojaRespuestas.getLastColumn();
  const encabezados = hojaRespuestas
    .getRange(1, 1, 1, ultimaColumna)
    .getDisplayValues()[0];

  const indice = (nombre) =>
    buscarIndice(encabezados, nombre);

  const idx = {
    id: buscarIndiceCualquiera(
      encabezados,
      ["ID_Rendicion", "ID"]
    ),
    colaborador: indice("colaborador"),
    rut: indice("rut"),
    numero_viaje: indice("numero_viaje"),
    fecha_documento: indice("fecha_documento"),
    numero_documento: indice("numero_documento"),
    proveedor: indice("proveedor"),
    monto_total: indice("monto_total"),
    descripcion: indice("descripcion"),
    fotografia: indice("FOTOGRAFIA"),
    archivo_url: indice("archivo_url"),
    estado_rendicion: indice("estado_rendicion"),
    autorizado_por: indice("autorizado_por"),
    autorizado_nombre: indice("autorizado_nombre"),
    autorizado_en: indice("autorizado_en"),
    id_comprobante: indice("id_comprobante"),
    estado_comprobante: indice("estado_comprobante"),
    revision_motivo: indice("revision_motivo"),
    revisado_por: indice("revisado_por"),
    revisado_nombre: indice("revisado_nombre"),
    revisado_en: indice("revisado_en")
  };

  const filasRendicion = buscarFilasPorValorExacto(
    hojaRespuestas,
    idx.id,
    id
  ).map((numeroFila) => ({
    fila: hojaRespuestas
      .getRange(numeroFila, 1, 1, ultimaColumna)
      .getDisplayValues()[0],
    numeroFila: numeroFila
  }));

  if (!filasRendicion.length) {
    return responderJSON({
      ok: false,
      error: "No se encontró la rendición.",
      ID: id
    });
  }

  const primeraFila =
    filasRendicion[0].fila;
  const colaborador = String(
    valorFila(
      primeraFila,
      idx.colaborador
    ) || ""
  ).trim();
  const rutOriginal =
    extraerRut(colaborador) ||
    valorFila(primeraFila, idx.rut);
  const rut = normalizarRutPermisos(
    rutOriginal
  );

  if (
    !esRolSuperAdmin(permisos) &&
    !usuarioTieneRutAsignado(
      permisos,
      rut
    )
  ) {
    return responderJSON({
      ok: false,
      error:
        "No puede revisar comprobantes de un colaborador no asignado."
    });
  }

  const datosColaborador =
    obtenerDatosColaborador(
      hojaColaboradores,
      rut
    );

  if (
    !empresaPermitidaParaConsulta(
      datosColaborador.empresa || "",
      permisos,
      empresaSeleccionada
    )
  ) {
    return responderJSON({
      ok: false,
      error:
        "No puede revisar comprobantes de esta empresa."
    });
  }

  const estadoActual = String(
    valorFila(
      primeraFila,
      idx.estado_rendicion
    ) || "PENDIENTE"
  ).trim().toUpperCase();

  if (estadoActual !== "PENDIENTE") {
    return responderJSON({
      ok: false,
      error:
        "La rendición ya fue finalizada y no puede revisarse nuevamente.",
      estado_rendicion: estadoActual
    });
  }

  const revisionesPorId = {};

  for (let i = 0; i < revisiones.length; i++) {
    const revision = revisiones[i] || {};
    const idComprobante = String(
      revision.id_comprobante || ""
    ).trim();
    const estado = String(
      revision.estado || ""
    ).trim().toUpperCase();
    const motivo = String(
      revision.motivo || ""
    ).trim();

    if (
      !idComprobante ||
      ![
        "AUTORIZADO",
        "RECHAZADO"
      ].includes(estado)
    ) {
      return responderJSON({
        ok: false,
        error:
          "Cada comprobante debe marcarse como autorizado o rechazado."
      });
    }

    if (
      estado === "RECHAZADO" &&
      !motivo
    ) {
      return responderJSON({
        ok: false,
        error:
          "Debe indicar el motivo de cada comprobante rechazado."
      });
    }

    revisionesPorId[idComprobante] = {
      estado: estado,
      motivo: motivo
    };
  }

  if (
    Object.keys(revisionesPorId).length !==
    filasRendicion.length
  ) {
    return responderJSON({
      ok: false,
      error:
        "Debe revisar todos los comprobantes de la rendición."
    });
  }

  for (let i = 0; i < filasRendicion.length; i++) {
    const idComprobante = String(
      valorFila(
        filasRendicion[i].fila,
        idx.id_comprobante
      ) || ""
    ).trim();

    if (!revisionesPorId[idComprobante]) {
      return responderJSON({
        ok: false,
        error:
          "La revisión no contiene todos los comprobantes de la rendición."
      });
    }
  }

  const ahora = new Date();
  let autorizados = 0;
  let rechazados = 0;
  let totalAutorizado = 0;
  let totalRechazado = 0;
  const comprobantesRechazados = [];

  filasRendicion.forEach((item) => {
    const idComprobante = String(
      valorFila(
        item.fila,
        idx.id_comprobante
      ) || ""
    ).trim();
    const revision =
      revisionesPorId[idComprobante];
    const monto = convertirNumero(
      valorFila(
        item.fila,
        idx.monto_total
      )
    );

    hojaRespuestas
      .getRange(
        item.numeroFila,
        idx.estado_comprobante + 1
      )
      .setValue(revision.estado);
    hojaRespuestas
      .getRange(
        item.numeroFila,
        idx.revision_motivo + 1
      )
      .setValue(revision.motivo);
    hojaRespuestas
      .getRange(
        item.numeroFila,
        idx.revisado_por + 1
      )
      .setValue(permisos.email || "");
    hojaRespuestas
      .getRange(
        item.numeroFila,
        idx.revisado_nombre + 1
      )
      .setValue(permisos.nombre || "");
    hojaRespuestas
      .getRange(
        item.numeroFila,
        idx.revisado_en + 1
      )
      .setValue(ahora);

    if (revision.estado === "AUTORIZADO") {
      autorizados++;
      totalAutorizado += monto;
    } else {
      rechazados++;
      totalRechazado += monto;
      comprobantesRechazados.push({
        id_comprobante: idComprobante,
        id_rendicion: id,
        colaborador: colaborador,
        rut: rutOriginal,
        empresa: datosColaborador.empresa || "",
        numero_viaje:
          valorFila(item.fila, idx.numero_viaje),
        fecha_documento:
          valorFila(item.fila, idx.fecha_documento),
        numero_documento:
          valorFila(item.fila, idx.numero_documento),
        proveedor:
          valorFila(item.fila, idx.proveedor),
        monto_total: monto,
        descripcion:
          valorFila(item.fila, idx.descripcion),
        motivo_rechazo: revision.motivo,
        fotografia:
          valorFila(item.fila, idx.fotografia) ||
          valorFila(item.fila, idx.archivo_url),
        revisado_por: permisos.email || "",
        revisado_nombre: permisos.nombre || "",
        revisado_en: ahora
      });
    }
  });

  let estadoRendicion = "PENDIENTE";

  if (autorizados === filasRendicion.length) {
    estadoRendicion = "AUTORIZADA";
  } else if (rechazados === filasRendicion.length) {
    estadoRendicion = "RECHAZADA";
  } else {
    estadoRendicion = "AUTORIZADA_PARCIAL";
  }

  filasRendicion.forEach((item) => {
    hojaRespuestas
      .getRange(
        item.numeroFila,
        idx.estado_rendicion + 1
      )
      .setValue(estadoRendicion);

    const tieneAutorizados = autorizados > 0;

    hojaRespuestas
      .getRange(
        item.numeroFila,
        idx.autorizado_por + 1
      )
      .setValue(
        tieneAutorizados
          ? permisos.email || ""
          : ""
      );
    hojaRespuestas
      .getRange(
        item.numeroFila,
        idx.autorizado_nombre + 1
      )
      .setValue(
        tieneAutorizados
          ? permisos.nombre || ""
          : ""
      );
    hojaRespuestas
      .getRange(
        item.numeroFila,
        idx.autorizado_en + 1
      )
      .setValue(
        tieneAutorizados
          ? ahora
          : ""
      );
  });

  if (comprobantesRechazados.length) {
    registrarComprobantesRechazados(
      comprobantesRechazados
    );
  }

  SpreadsheetApp.flush();
  invalidarCacheListadoRendiciones();

  return responderJSON({
    ok: true,
    ID: id,
    estado_rendicion: estadoRendicion,
    comprobantes_autorizados: autorizados,
    comprobantes_rechazados: rechazados,
    total_autorizado: totalAutorizado,
    total_rechazado: totalRechazado,
    revisado_por: permisos.email || "",
    revisado_nombre: permisos.nombre || "",
    revisado_en: ahora.toISOString()
  });
}


function registrarComprobantesRechazados(
  comprobantes
) {
  const libro = obtenerLibroRendiciones();
  let hoja = libro.getSheetByName(
    "COMPROBANTES_RECHAZADOS"
  );

  const encabezados = [
    "id_comprobante",
    "id_rendicion",
    "colaborador",
    "rut",
    "empresa",
    "numero_viaje",
    "fecha_documento",
    "numero_documento",
    "proveedor",
    "monto_total",
    "descripcion",
    "motivo_rechazo",
    "fotografia",
    "revisado_por",
    "revisado_nombre",
    "revisado_en"
  ];

  if (!hoja) {
    hoja = libro.insertSheet(
      "COMPROBANTES_RECHAZADOS"
    );
    hoja
      .getRange(
        1,
        1,
        1,
        encabezados.length
      )
      .setValues([encabezados]);
  }

  const filas = comprobantes.map(
    (item) =>
      encabezados.map(
        (encabezado) =>
          item[encabezado] === undefined
            ? ""
            : item[encabezado]
      )
  );

  hoja
    .getRange(
      hoja.getLastRow() + 1,
      1,
      filas.length,
      encabezados.length
    )
    .setValues(filas);
}

function obtenerListadoComprobantesRechazados(
  hoja,
  permisos,
  empresaSeleccionada,
  parametros
) {
  // La hoja se crea al registrar el primer rechazo. Si todavía
  // no existe, el reporte debe responder una lista vacía.
  if (!hoja || hoja.getLastRow() < 2) {
    return responderJSON([]);
  }

  const rango = hoja.getDataRange();
  const valores = rango.getValues();
  const valoresMostrados = rango.getDisplayValues();
  const encabezados = valoresMostrados[0];

  const indice = (nombre) =>
    buscarIndice(encabezados, nombre);

  const idx = {
    id_comprobante: indice("id_comprobante"),
    id_rendicion: indice("id_rendicion"),
    colaborador: indice("colaborador"),
    rut: indice("rut"),
    empresa: indice("empresa"),
    numero_viaje: indice("numero_viaje"),
    fecha_documento: indice("fecha_documento"),
    numero_documento: indice("numero_documento"),
    proveedor: indice("proveedor"),
    monto_total: indice("monto_total"),
    descripcion: indice("descripcion"),
    motivo_rechazo: indice("motivo_rechazo"),
    fotografia: indice("fotografia"),
    revisado_por: indice("revisado_por"),
    revisado_nombre: indice("revisado_nombre"),
    revisado_en: indice("revisado_en")
  };

  const columnasObligatorias = [
    "id_comprobante",
    "id_rendicion",
    "colaborador",
    "rut",
    "empresa",
    "monto_total",
    "motivo_rechazo",
    "revisado_en"
  ];

  const faltantes = columnasObligatorias.filter(
    (nombre) => idx[nombre] < 0
  );

  if (faltantes.length) {
    return responderJSON({
      error:
        "La hoja COMPROBANTES_RECHAZADOS no contiene todas las columnas requeridas.",
      columnas_faltantes: faltantes
    });
  }

  const rutSolicitado = normalizarRutPermisos(
    parametros.rut || ""
  );
  const fechaDesde = crearLimiteFechaReporte(
    parametros.fecha_desde || "",
    false
  );
  const fechaHasta = crearLimiteFechaReporte(
    parametros.fecha_hasta || "",
    true
  );
  const resultados = [];

  for (let i = 1; i < valoresMostrados.length; i++) {
    const fila = valoresMostrados[i];
    const filaReal = valores[i];
    const rutOriginal = String(
      valorFila(fila, idx.rut) ||
      extraerRut(
        valorFila(fila, idx.colaborador)
      ) ||
      ""
    ).trim();
    const rutNormalizado = normalizarRutPermisos(
      rutOriginal
    );
    const empresa = String(
      valorFila(fila, idx.empresa) || ""
    ).trim();

    if (!rutNormalizado) {
      continue;
    }

    if (
      !esRolSuperAdmin(permisos) &&
      !usuarioTieneRutAsignado(
        permisos,
        rutNormalizado
      )
    ) {
      continue;
    }

    if (
      !empresaPermitidaParaConsulta(
        empresa,
        permisos,
        empresaSeleccionada
      )
    ) {
      continue;
    }

    if (
      rutSolicitado &&
      rutNormalizado !== rutSolicitado
    ) {
      continue;
    }

    const fechaRevisionReal =
      valorFila(filaReal, idx.revisado_en);
    const fechaRevision = convertirFechaReporte(
      fechaRevisionReal ||
      valorFila(fila, idx.revisado_en)
    );

    if (
      fechaDesde &&
      (!fechaRevision || fechaRevision < fechaDesde)
    ) {
      continue;
    }

    if (
      fechaHasta &&
      (!fechaRevision || fechaRevision > fechaHasta)
    ) {
      continue;
    }

    resultados.push({
      id_comprobante: String(
        valorFila(fila, idx.id_comprobante) || ""
      ).trim(),
      id_rendicion: String(
        valorFila(fila, idx.id_rendicion) || ""
      ).trim(),
      colaborador: String(
        valorFila(fila, idx.colaborador) || ""
      ).trim(),
      rut: rutOriginal,
      empresa: empresa,
      numero_viaje: String(
        valorFila(fila, idx.numero_viaje) || ""
      ).trim(),
      fecha_documento: String(
        valorFila(fila, idx.fecha_documento) || ""
      ).trim(),
      numero_documento: String(
        valorFila(fila, idx.numero_documento) || ""
      ).trim(),
      proveedor: String(
        valorFila(fila, idx.proveedor) || ""
      ).trim(),
      monto_total: convertirNumero(
        valorFila(fila, idx.monto_total)
      ),
      descripcion: String(
        valorFila(fila, idx.descripcion) || ""
      ).trim(),
      motivo_rechazo: String(
        valorFila(fila, idx.motivo_rechazo) || ""
      ).trim(),
      fotografia: String(
        valorFila(fila, idx.fotografia) || ""
      ).trim(),
      revisado_por: String(
        valorFila(fila, idx.revisado_por) || ""
      ).trim(),
      revisado_nombre: String(
        valorFila(fila, idx.revisado_nombre) || ""
      ).trim(),
      revisado_en:
        fechaRevision &&
        !isNaN(fechaRevision.getTime())
          ? fechaRevision.toISOString()
          : String(
              valorFila(fila, idx.revisado_en) || ""
            ).trim(),
      estado_comprobante: "RECHAZADO",
      _orden_fecha:
        fechaRevision &&
        !isNaN(fechaRevision.getTime())
          ? fechaRevision.getTime()
          : 0
    });
  }

  resultados.sort(
    (a, b) =>
      b._orden_fecha - a._orden_fecha ||
      String(b.id_rendicion).localeCompare(
        String(a.id_rendicion)
      )
  );

  resultados.forEach((item) => {
    delete item._orden_fecha;
  });

  return responderJSON(resultados);
}


function crearLimiteFechaReporte(
  valor,
  finDelDia
) {
  const texto = String(valor || "").trim();

  if (!texto) {
    return null;
  }

  const partes = texto.match(
    /^(\d{4})-(\d{2})-(\d{2})$/
  );

  if (!partes) {
    return null;
  }

  return new Date(
    Number(partes[1]),
    Number(partes[2]) - 1,
    Number(partes[3]),
    finDelDia ? 23 : 0,
    finDelDia ? 59 : 0,
    finDelDia ? 59 : 0,
    finDelDia ? 999 : 0
  );
}


function convertirFechaReporte(valor) {
  if (
    valor instanceof Date &&
    !isNaN(valor.getTime())
  ) {
    return valor;
  }

  const texto = String(valor || "").trim();

  if (!texto) {
    return null;
  }

  const chilena = texto.match(
    /^(\d{1,2})[\/-](\d{1,2})[\/-](\d{4})(?:\s+(\d{1,2}):(\d{2})(?::(\d{2}))?)?$/
  );

  if (chilena) {
    const fecha = new Date(
      Number(chilena[3]),
      Number(chilena[2]) - 1,
      Number(chilena[1]),
      Number(chilena[4] || 0),
      Number(chilena[5] || 0),
      Number(chilena[6] || 0)
    );

    return isNaN(fecha.getTime())
      ? null
      : fecha;
  }

  const fecha = new Date(texto);

  return isNaN(fecha.getTime())
    ? null
    : fecha;
}


function usuarioTieneRutAsignado(
  permisos,
  rut
) {
  const rutNormalizado =
    normalizarRutPermisos(rut);

  if (!rutNormalizado) {
    return false;
  }

  const asignados =
    permisos &&
    Array.isArray(permisos.colaboradores)
      ? permisos.colaboradores.map(
          normalizarRutPermisos
        )
      : [];

  return asignados.includes(
    rutNormalizado
  );
}


function normalizarEmpresaSistema(empresa) {
  const valor = String(empresa || "")
    .trim()
    .toUpperCase()
    .replace(/\s+/g, " ");

  if (
    valor === "SERVICIOS INDUSTRIALES" ||
    valor === "SERVICIOS INDUSTRIALES SPA"
  ) {
    return "SERVIND";
  }

  return valor;
}


function empresaPermitidaParaConsulta(
  empresaDato,
  permisos,
  empresaSeleccionada
) {
  const empresaRegistro =
    normalizarEmpresaSistema(
      empresaDato
    );

  if (esRolSuperAdmin(permisos)) {
    const seleccion =
      normalizarEmpresaSistema(
        empresaSeleccionada
      );

    return (
      !seleccion ||
      seleccion === "TODAS" ||
      seleccion === empresaRegistro
    );
  }

  const empresaUsuario =
    normalizarEmpresaSistema(
      permisos && permisos.empresa
    );

  return (
    !empresaUsuario ||
    !empresaRegistro ||
    empresaUsuario === empresaRegistro
  );
}

function obtenerDatosHojaConsultaCached(hoja, clave) {
  if (!hoja) {
    return [];
  }

  if (!obtenerDatosHojaConsultaCached._cache) {
    obtenerDatosHojaConsultaCached._cache = {};

    return obtenerDatosHojaConsultaCached._cache[cacheKey];
  }

  const datos = hoja.getDataRange().getDisplayValues();
  obtenerDatosHojaConsultaCached._cache[cacheKey] = datos;
  return datos;
}


function buscarFilasPorValorExacto(hoja, indiceColumna, valor) {
  if (!hoja || indiceColumna < 0 || !String(valor || "").trim()) {
    return [];
  }

  const ultimaFila = hoja.getLastRow();
  if (ultimaFila < 2) {
    return [];
  }

  return hoja
    .getRange(2, indiceColumna + 1, ultimaFila - 1, 1)
    .createTextFinder(String(valor).trim())
    .matchEntireCell(true)
    .useRegularExpression(false)
    .findAll()
    .map((celda) => celda.getRow());
}

function obtenerClaveCacheConsulta(tipo, permisos, empresaSeleccionada) {
  if (!permisos || !permisos.email) {
    return "";
  }

  const contenido = [
    String(tipo || "CONSULTA").trim().toUpperCase(),
    String(permisos.email || "").trim().toLowerCase(),
    String(permisos.rol || "").trim().toUpperCase(),
    String(empresaSeleccionada || "").trim().toUpperCase(),
    obtenerVersionCacheConsultas()
  ].join("|");

  const digest = Utilities.computeDigest(
    Utilities.DigestAlgorithm.MD5,
    contenido
  );

  return "CONSULTA_" +
    Utilities.base64EncodeWebSafe(digest).replace(/=+$/g, "");
}

  );
}


function leerCacheJSON(cacheKey) {
  if (!cacheKey) {
    return null;
  }

  try {
    const cache = CacheService.getScriptCache();
    const meta = cache.get(cacheKey + "_META");

    if (!meta) {
      const directo = cache.get(cacheKey);
      return directo ? JSON.parse(directo) : null;
    }

    const cantidad = Number(meta);
    if (!Number.isFinite(cantidad) || cantidad < 1 || cantidad > 20) {
      return null;
    }

    const claves = [];
    for (let i = 0; i < cantidad; i++) {
      claves.push(cacheKey + "_" + i);
    }

    const partes = cache.getAll(claves);
    const base64 = claves.map((clave) => partes[clave] || "").join("");
    if (!base64) {
      return null;
    }

    const bytes = Utilities.base64Decode(base64);
    const texto = Utilities
      .ungzip(Utilities.newBlob(bytes))
      .getDataAsString();
    return JSON.parse(texto);
  } catch (error) {
    console.warn("No fue posible leer la caché de consulta:", error.message);
    return null;
  }
}


function guardarCacheJSON(cacheKey, valor, segundos) {
  if (!cacheKey) {
    return;
  }

  try {
    const cache = CacheService.getScriptCache();
    const texto = JSON.stringify(valor);

    // Primero intentamos guardar directamente para respuestas pequeñas.
    if (texto.length <= 90000) {
      cache.put(cacheKey, texto, segundos);
      return;
    }

    // Para listados grandes comprimimos y dividimos para respetar el límite
    // de tamaño de una entrada de CacheService.
    const comprimido = Utilities.gzip(
      Utilities.newBlob(texto, "application/json", "consulta.json")
    );
    const base64 = Utilities.base64Encode(comprimido.getBytes());
    const tamanoParte = 80000;
    const cantidad = Math.ceil(base64.length / tamanoParte);

    if (cantidad > 20) {
      return;
    }

    const partes = {};
    for (let i = 0; i < cantidad; i++) {
      partes[cacheKey + "_" + i] = base64.slice(
        i * tamanoParte,
        (i + 1) * tamanoParte
      );
    }

    cache.putAll(partes, segundos);
    cache.put(cacheKey + "_META", String(cantidad), segundos);
  } catch (error) {
    console.warn("No fue posible guardar la caché de consulta:", error.message);
  }
}

function obtenerEmpresasPorRut(
  hojaColaboradores
) {
  if (obtenerEmpresasPorRut._cache) {
    return obtenerEmpresasPorRut._cache;
  }

  const resultado = {};

  if (
    !hojaColaboradores ||
    hojaColaboradores.getLastRow() < 2
  ) {
    return resultado;
  }

  const datos = hojaColaboradores
    .getDataRange()
    .getDisplayValues();

  const encabezados = datos[0];
  const idxRut = buscarIndice(
    encabezados,
    "rut"
  );
  const idxEmpresa = buscarIndice(
    encabezados,
    "empresa"
  );

  if (idxRut < 0 || idxEmpresa < 0) {
    return resultado;
  }

  for (let i = 1; i < datos.length; i++) {
    const rut = normalizarRutPermisos(
      valorFila(datos[i], idxRut)
    );

    if (rut) {
      resultado[rut] = String(
        valorFila(datos[i], idxEmpresa) || ""
      ).trim();
    }
  }

  obtenerEmpresasPorRut._cache = resultado;
  return resultado;
}

function obtenerDatosColaborador(
  hoja,
  rutBuscado
) {
  if (!hoja || !rutBuscado) {
    return {
      empresa: "",
      centro_costo: ""
    };
  }

  const rutObjetivo =
    normalizarRutPermisos(rutBuscado);

  if (!rutObjetivo) {
    return {
      empresa: "",
      centro_costo: ""
    };
  }

  const sheetId = hoja.getSheetId();

  if (
    !obtenerDatosColaborador._cache ||
    obtenerDatosColaborador._cache.sheetId !== sheetId
  ) {
    const ultimaFila = hoja.getLastRow();
    const mapa = {};

    if (ultimaFila >= 2) {
      const datos = hoja
        .getRange(2, 1, ultimaFila - 1, 4)
        .getDisplayValues();

      datos.forEach((fila) => {
        const rut = normalizarRutPermisos(fila[0]);
        if (!rut || mapa[rut]) {
          return;
        }

        mapa[rut] = {
          centro_costo: fila[2] || "",
          empresa: fila[3] || ""
        };
      });
    }

    obtenerDatosColaborador._cache = {
      sheetId: sheetId,
      mapa: mapa
    };
  }

  return (
    obtenerDatosColaborador._cache.mapa[rutObjetivo] || {
      empresa: "",
      centro_costo: ""
    }
  );
}

// HOJA CONSOLIDADO
// ======================================================

function obtenerDatosConsolidado(
  hoja,
  rutBuscado,
  colaboradorCompleto
) {
  if (!hoja) {
    return {
      monto_asignado: 0,
      saldo_2: 0
    };
  }

  const ultimaFila =
    hoja.getLastRow();

  if (ultimaFila < 3) {
    return {
      monto_asignado: 0,
      saldo_2: 0
    };
  }

  const datos = hoja
    .getRange(
      3,
      1,
      ultimaFila - 2,
      4
    )
    .getDisplayValues();

  for (
    let i = 0;
    i < datos.length;
    i++
  ) {
    const colaborador =
      String(
        datos[i][0] || ""
      ).trim();

    if (!colaborador) {
      continue;
    }

    const rutColaborador =
      extraerRut(
        colaborador
      );

    const coincideRut =
      normalizarRutPermisos(
        rutBuscado
      ) &&
      normalizarRutPermisos(
        rutBuscado
      ) ===
        normalizarRutPermisos(
          rutColaborador
        );

    const coincideNombre =
      colaboradorCompleto &&
      colaboradorCompleto.includes(
        colaborador
      );

    if (
      coincideRut ||
      coincideNombre
    ) {
      return {
        monto_asignado:
          convertirNumero(
            datos[i][1]
          ),

        saldo_2:
          convertirNumero(
            datos[i][3]
          )
      };
    }
  }

  return {
    monto_asignado: 0,
    saldo_2: 0
  };
}

function asegurarEstructuraAbonos(hojaAbonos) {
  if (!hojaAbonos) {
    throw new Error(
      'No existe la hoja "ABONOS".'
    );
  }

  const ultimaColumna = Math.max(
    hojaAbonos.getLastColumn(),
    3
  );

  let encabezados = hojaAbonos
    .getRange(1, 1, 1, ultimaColumna)
    .getDisplayValues()[0]
    .map((valor) => String(valor || "").trim());

  [
    "COLABORADOR",
    "FECHA",
    "DEPOSITO"
  ].forEach((nombre) => {
    if (buscarIndice(encabezados, nombre) < 0) {
      throw new Error(
        'La hoja ABONOS debe conservar las columnas COLABORADOR, FECHA y DEPOSITO.'
      );
    }
  });

  let modificada = false;

  COLUMNAS_ABONOS_AUDITORIA.forEach((nombre) => {
    if (buscarIndice(encabezados, nombre) < 0) {
      encabezados.push(nombre);
      modificada = true;
    }
  });

  if (modificada) {
    hojaAbonos
      .getRange(1, 1, 1, encabezados.length)
      .setValues([encabezados]);
  }

  const indices = {};

  encabezados.forEach((nombre, indice) => {
    indices[String(nombre || "")
      .trim()
      .toUpperCase()] = indice;
  });

  return {
    encabezados: encabezados,
    indices: indices
  };
}


function obtenerColaboradorAbonos(
  hojaColaboradores,
  rutBuscado
) {
  const rutNormalizado =
    normalizarRutPermisos(rutBuscado);

  if (
    !hojaColaboradores ||
    !rutNormalizado ||
    hojaColaboradores.getLastRow() < 2
  ) {
    return null;
  }

  const datos = hojaColaboradores
    .getDataRange()
    .getDisplayValues();

  const encabezados = datos[0];
  const idxRut = buscarIndice(
    encabezados,
    "rut"
  );
  const idxColaborador = buscarIndice(
    encabezados,
    "colaborador"
  );
  const idxEmpresa = buscarIndice(
    encabezados,
    "empresa"
  );
  const idxCentroCosto = buscarIndice(
    encabezados,
    "centro de costo"
  );

  if (
    idxRut < 0 ||
    idxColaborador < 0 ||
    idxEmpresa < 0
  ) {
    throw new Error(
      "COLABORADORES debe contener rut, colaborador y empresa."
    );
  }

  for (let i = 1; i < datos.length; i++) {
    if (
      normalizarRutPermisos(
        valorFila(datos[i], idxRut)
      ) === rutNormalizado
    ) {
      return {
        rut: String(
          valorFila(datos[i], idxRut) || ""
        ).trim(),
        colaborador: String(
          valorFila(datos[i], idxColaborador) || ""
        ).trim(),
        empresa: String(
          valorFila(datos[i], idxEmpresa) || ""
        ).trim(),
        centro_costo:
          idxCentroCosto >= 0
            ? String(
                valorFila(
                  datos[i],
                  idxCentroCosto
                ) || ""
              ).trim()
            : ""
      };
    }
  }

  return null;
}


function validarAccesoColaboradorAbonos(
  colaborador,
  permisos,
  empresaSeleccionada
) {
  if (!colaborador) {
    return false;
  }

  if (
    !esRolSuperAdmin(permisos) &&
    !usuarioTieneRutAsignado(
      permisos,
      colaborador.rut
    )
  ) {
    return false;
  }

  return empresaPermitidaParaConsulta(
    colaborador.empresa,
    permisos,
    empresaSeleccionada
  );
}


function generarIdAbono() {
  const zona = Session.getScriptTimeZone() ||
    "America/Santiago";
  const fecha = Utilities.formatDate(
    new Date(),
    zona,
    "yyyyMMdd-HHmmss"
  );
  const aleatorio = Utilities
    .getUuid()
    .replace(/-/g, "")
    .substring(0, 6)
    .toUpperCase();

  return "AB-" + fecha + "-" + aleatorio;
}


function registrarAbonoDesdeGet(
  hojaAbonos,
  hojaColaboradores,
  parametros,
  permisos,
  empresaSeleccionada
) {
  if (!esRolAutorizador(permisos)) {
    return responderJSON({
      error:
        "Solo el Jefe de Área o Super Administrador puede registrar abonos."
    });
  }

  const rut = String(
    parametros.rut || ""
  ).trim();
  const monto = convertirNumero(
    parametros.monto
  );
  const fecha = String(
    parametros.fecha || ""
  ).trim();

  if (!rut) {
    return responderJSON({
      error:
        "Debe seleccionar un colaborador."
    });
  }

  if (!(monto > 0)) {
    return responderJSON({
      error:
        "El monto del abono debe ser mayor que cero."
    });
  }

  if (!fecha) {
    return responderJSON({
      error:
        "Debe indicar la fecha del abono."
    });
  }

  const colaborador = obtenerColaboradorAbonos(
    hojaColaboradores,
    rut
  );

  if (!colaborador) {
    return responderJSON({
      error:
        "El colaborador no existe en la hoja COLABORADORES."
    });
  }

  if (
    !validarAccesoColaboradorAbonos(
      colaborador,
      permisos,
      empresaSeleccionada
    )
  ) {
    return responderJSON({
      error:
        "No tiene permisos para registrar abonos a este colaborador."
    });
  }

  const bloqueo = LockService.getScriptLock();

  try {
    bloqueo.waitLock(30000);

    const estructura = asegurarEstructuraAbonos(
      hojaAbonos
    );
    const fila = new Array(
      estructura.encabezados.length
    ).fill("");
    const idx = estructura.indices;
    const ahora = new Date();
    const idAbono = generarIdAbono();

    fila[idx.COLABORADOR] =
      colaborador.colaborador;
    fila[idx.FECHA] = fecha;
    fila[idx.DEPOSITO] = monto;
    fila[idx.ID_ABONO] = idAbono;
    fila[idx.EMPRESA] = colaborador.empresa;
    fila[idx.MEDIO_PAGO] = String(
      parametros.medio_pago || ""
    ).trim();
    fila[idx.REFERENCIA] = String(
      parametros.referencia || ""
    ).trim();
    fila[idx.OBSERVACION] = String(
      parametros.observacion || ""
    ).trim();
    fila[idx.COMPROBANTE_URL] = String(
      parametros.comprobante_url || ""
    ).trim();
    fila[idx.USUARIO_REGISTRO] =
      permisos.email || "";
    fila[idx.NOMBRE_REGISTRO] =
      permisos.nombre || "";
    fila[idx.REGISTRADO_EN] = ahora;
    fila[idx.ESTADO] = "ACTIVO";

    hojaAbonos
      .getRange(
        hojaAbonos.getLastRow() + 1,
        1,
        1,
        fila.length
      )
      .setValues([fila]);

    SpreadsheetApp.flush();

    return responderJSON({
      ok: true,
      mensaje: "Abono registrado correctamente.",
      id_abono: idAbono,
      rut: colaborador.rut,
      colaborador: colaborador.colaborador,
      empresa: colaborador.empresa,
      fecha: fecha,
      monto: monto,
      estado: "ACTIVO"
    });
  } catch (error) {
    return responderJSON({
      error:
        "No fue posible registrar el abono: " +
        error.message
    });
  } finally {
    try {
      bloqueo.releaseLock();
    } catch (errorBloqueo) {
      // El bloqueo pudo expirar antes de liberarse.
    }
  }
}


function obtenerListadoAbonos(
  hojaAbonos,
  hojaColaboradores,
  permisos,
  empresaSeleccionada,
  parametros
) {
  if (!hojaAbonos) {
    return responderJSON({
      error: 'No existe la hoja "ABONOS".'
    });
  }

  const estructura = asegurarEstructuraAbonos(
    hojaAbonos
  );
  const idx = estructura.indices;
  const datos = hojaAbonos
    .getRange(
      1,
      1,
      Math.max(hojaAbonos.getLastRow(), 1),
      estructura.encabezados.length
    )
    .getDisplayValues();
  const filtroRut = normalizarRutPermisos(
    parametros.rut || ""
  );
  const filtroEstado = String(
    parametros.estado || ""
  ).trim().toUpperCase();
  const resultado = [];

  for (let i = 1; i < datos.length; i++) {
    const colaboradorTexto = String(
      valorFila(datos[i], idx.COLABORADOR) || ""
    ).trim();

    if (!colaboradorTexto) {
      continue;
    }

    const rut = extraerRut(colaboradorTexto);
    const colaborador = obtenerColaboradorAbonos(
      hojaColaboradores,
      rut
    );

    if (
      !colaborador ||
      !validarAccesoColaboradorAbonos(
        colaborador,
        permisos,
        empresaSeleccionada
      )
    ) {
      continue;
    }

    if (
      filtroRut &&
      filtroRut !== normalizarRutPermisos(rut)
    ) {
      continue;
    }

    const estado = String(
      valorFila(datos[i], idx.ESTADO) ||
      "ACTIVO"
    ).trim().toUpperCase();

    if (
      filtroEstado &&
      filtroEstado !== "TODOS" &&
      filtroEstado !== estado
    ) {
      continue;
    }

    const idReal = String(
      valorFila(datos[i], idx.ID_ABONO) || ""
    ).trim();

    const montoAbono = convertirNumero(
      valorFila(datos[i], idx.DEPOSITO)
    );

    // Las filas antiguas sin ID y sin depósito no representan un
    // movimiento financiero. Se conservan en la planilla, pero no se
    // muestran como abonos históricos en el sistema.
    if (!idReal && montoAbono === 0) {
      continue;
    }

    resultado.push({
      id_abono:
        idReal ||
        "LEGACY-" + String(i + 1),
      es_legacy: !idReal,
      anulable: Boolean(idReal) &&
        estado === "ACTIVO",
      rut: colaborador.rut,
      colaborador: colaborador.colaborador,
      empresa:
        String(
          valorFila(datos[i], idx.EMPRESA) ||
          colaborador.empresa
        ).trim(),
      fecha: valorFila(datos[i], idx.FECHA),
      monto: montoAbono,
      medio_pago: valorFila(
        datos[i],
        idx.MEDIO_PAGO
      ),
      referencia: valorFila(
        datos[i],
        idx.REFERENCIA
      ),
      observacion: valorFila(
        datos[i],
        idx.OBSERVACION
      ),
      comprobante_url: valorFila(
        datos[i],
        idx.COMPROBANTE_URL
      ),
      usuario_registro: valorFila(
        datos[i],
        idx.USUARIO_REGISTRO
      ),
      nombre_registro: valorFila(
        datos[i],
        idx.NOMBRE_REGISTRO
      ),
      registrado_en: valorFila(
        datos[i],
        idx.REGISTRADO_EN
      ),
      estado: estado,
      anulado_por: valorFila(
        datos[i],
        idx.ANULADO_POR
      ),
      anulado_nombre: valorFila(
        datos[i],
        idx.ANULADO_NOMBRE
      ),
      anulado_en: valorFila(
        datos[i],
        idx.ANULADO_EN
      ),
      motivo_anulacion: valorFila(
        datos[i],
        idx.MOTIVO_ANULACION
      )
    });
  }

  resultado.sort((a, b) => {
    const fechaA = convertirFechaReporte(
      a.registrado_en || a.fecha
    );
    const fechaB = convertirFechaReporte(
      b.registrado_en || b.fecha
    );

    return (
      (fechaB ? fechaB.getTime() : 0) -
      (fechaA ? fechaA.getTime() : 0)
    );
  });

  return responderJSON(resultado);
}


function anularAbonoDesdeGet(
  hojaAbonos,
  hojaColaboradores,
  parametros,
  permisos,
  empresaSeleccionada
) {
  if (!esRolAutorizador(permisos)) {
    return responderJSON({
      error:
        "Solo el Jefe de Área o Super Administrador puede anular abonos."
    });
  }

  const idAbono = String(
    parametros.id_abono || ""
  ).trim();
  const motivo = String(
    parametros.motivo || ""
  ).trim();

  if (!idAbono) {
    return responderJSON({
      error: "Debe indicar el ID del abono."
    });
  }

  if (!motivo) {
    return responderJSON({
      error:
        "Debe indicar el motivo de la anulación."
    });
  }

  const bloqueo = LockService.getScriptLock();

  try {
    bloqueo.waitLock(30000);

    const estructura = asegurarEstructuraAbonos(
      hojaAbonos
    );
    const idx = estructura.indices;
    const ultimaFila = hojaAbonos.getLastRow();

    if (ultimaFila < 2) {
      return responderJSON({
        error: "No existen abonos registrados."
      });
    }

    const datos = hojaAbonos
      .getRange(
        2,
        1,
        ultimaFila - 1,
        estructura.encabezados.length
      )
      .getDisplayValues();

    for (let i = 0; i < datos.length; i++) {
      if (
        String(
          valorFila(datos[i], idx.ID_ABONO) || ""
        ).trim() !== idAbono
      ) {
        continue;
      }

      const colaborador = obtenerColaboradorAbonos(
        hojaColaboradores,
        extraerRut(
          valorFila(datos[i], idx.COLABORADOR)
        )
      );

      if (
        !validarAccesoColaboradorAbonos(
          colaborador,
          permisos,
          empresaSeleccionada
        )
      ) {
        return responderJSON({
          error:
            "No tiene permisos para anular este abono."
        });
      }

      const estadoActual = String(
        valorFila(datos[i], idx.ESTADO) ||
        "ACTIVO"
      ).trim().toUpperCase();

      if (estadoActual === "ANULADO") {
        return responderJSON({
          error: "El abono ya se encuentra anulado."
        });
      }

      const numeroFila = i + 2;
      hojaAbonos.getRange(
        numeroFila,
        idx.ESTADO + 1
      ).setValue("ANULADO");
      hojaAbonos.getRange(
        numeroFila,
        idx.ANULADO_POR + 1
      ).setValue(permisos.email || "");
      hojaAbonos.getRange(
        numeroFila,
        idx.ANULADO_NOMBRE + 1
      ).setValue(permisos.nombre || "");
      hojaAbonos.getRange(
        numeroFila,
        idx.ANULADO_EN + 1
      ).setValue(new Date());
      hojaAbonos.getRange(
        numeroFila,
        idx.MOTIVO_ANULACION + 1
      ).setValue(motivo);

      SpreadsheetApp.flush();

      return responderJSON({
        ok: true,
        mensaje: "Abono anulado correctamente.",
        id_abono: idAbono,
        estado: "ANULADO"
      });
    }

    return responderJSON({
      error: "No se encontró el abono solicitado."
    });
  } catch (error) {
    return responderJSON({
      error:
        "No fue posible anular el abono: " +
        error.message
    });
  } finally {
    try {
      bloqueo.releaseLock();
    } catch (errorBloqueo) {
      // El bloqueo pudo expirar antes de liberarse.
    }
  }
}


function obtenerCartolaColaborador(
  hojaAbonos,
  hojaRespuestas,
  hojaColaboradores,
  permisos,
  empresaSeleccionada,
  parametros
) {
  const rut = String(
    parametros.rut || ""
  ).trim();
  const colaborador = obtenerColaboradorAbonos(
    hojaColaboradores,
    rut
  );

  if (!colaborador) {
    return responderJSON({
      error:
        "Debe indicar un colaborador válido."
    });
  }

  if (
    !validarAccesoColaboradorAbonos(
      colaborador,
      permisos,
      empresaSeleccionada
    )
  ) {
    return responderJSON({
      error:
        "No tiene permisos para consultar esta cartola."
    });
  }

  const movimientos = [];
  let totalAbonos = 0;
  let totalRendido = 0;

  if (hojaAbonos && hojaAbonos.getLastRow() >= 2) {
    const estructura = asegurarEstructuraAbonos(
      hojaAbonos
    );
    const idx = estructura.indices;
    const datos = hojaAbonos
      .getRange(
        2,
        1,
        hojaAbonos.getLastRow() - 1,
        estructura.encabezados.length
      )
      .getDisplayValues();

    datos.forEach((fila, indice) => {
      const rutFila = normalizarRutPermisos(
        extraerRut(
          valorFila(fila, idx.COLABORADOR)
        )
      );
      const estado = String(
        valorFila(fila, idx.ESTADO) ||
        "ACTIVO"
      ).trim().toUpperCase();

      if (
        rutFila !== normalizarRutPermisos(rut) ||
        estado === "ANULADO"
      ) {
        return;
      }

      const monto = convertirNumero(
        valorFila(fila, idx.DEPOSITO)
      );
      totalAbonos += monto;

      movimientos.push({
        id_movimiento:
          String(
            valorFila(fila, idx.ID_ABONO) || ""
          ).trim() ||
          "LEGACY-" + String(indice + 2),
        fecha: valorFila(fila, idx.FECHA),
        fecha_orden:
          valorFila(fila, idx.REGISTRADO_EN) ||
          valorFila(fila, idx.FECHA),
        tipo: "ABONO",
        detalle:
          valorFila(fila, idx.OBSERVACION) ||
          "Abono de fondos",
        referencia: valorFila(
          fila,
          idx.REFERENCIA
        ),
        debe: 0,
        haber: monto,
        monto: monto,
        comprobante_url: valorFila(
          fila,
          idx.COMPROBANTE_URL
        )
      });
    });
  }

  if (
    hojaRespuestas &&
    hojaRespuestas.getLastRow() >= 2
  ) {
    const datos = hojaRespuestas
      .getDataRange()
      .getDisplayValues();
    const encabezados = datos[0];
    const idxColaborador = buscarIndice(
      encabezados,
      "colaborador"
    );
    const idxRut = buscarIndice(
      encabezados,
      "rut"
    );
    const idxId = buscarIndiceCualquiera(
      encabezados,
      ["ID_Rendicion", "ID"]
    );
    const idxMonto = buscarIndiceCualquiera(
      encabezados,
      ["monto_total", "MONTO"]
    );
    const idxProcesado = buscarIndice(
      encabezados,
      "procesado_en"
    );
    const idxEstadoComprobante = buscarIndice(
      encabezados,
      "estado_comprobante"
    );
    const idxEstadoRendicion = buscarIndice(
      encabezados,
      "estado_rendicion"
    );
    const porRendicion = {};

    datos.slice(1).forEach((fila) => {
      const rutFila = normalizarRutPermisos(
        extraerRut(
          valorFila(fila, idxColaborador)
        ) || valorFila(fila, idxRut)
      );

      if (rutFila !== normalizarRutPermisos(rut)) {
        return;
      }

      const estadoComprobante =
        idxEstadoComprobante >= 0
          ? String(
              valorFila(
                fila,
                idxEstadoComprobante
              ) || ""
            ).trim().toUpperCase()
          : "";
      const estadoRendicion =
        idxEstadoRendicion >= 0
          ? String(
              valorFila(
                fila,
                idxEstadoRendicion
              ) || "PENDIENTE"
            ).trim().toUpperCase()
          : "PENDIENTE";
      const autorizado =
        estadoComprobante === "AUTORIZADO" ||
        (
          !estadoComprobante &&
          estadoRendicion === "AUTORIZADA"
        );

      if (!autorizado) {
        return;
      }

      const id = String(
        valorFila(fila, idxId) || ""
      ).trim();

      if (!id) {
        return;
      }

      if (!porRendicion[id]) {
        porRendicion[id] = {
          monto: 0,
          fecha:
            idxProcesado >= 0
              ? valorFila(fila, idxProcesado)
              : ""
        };
      }

      porRendicion[id].monto += convertirNumero(
        valorFila(fila, idxMonto)
      );
    });

    Object.keys(porRendicion).forEach((id) => {
      const item = porRendicion[id];
      totalRendido += item.monto;
      movimientos.push({
        id_movimiento: id,
        fecha: item.fecha,
        fecha_orden: item.fecha,
        tipo: "RENDICION",
        detalle: "Rendición autorizada " + id,
        referencia: id,
        debe: item.monto,
        haber: 0,
        monto: -item.monto,
        comprobante_url: ""
      });
    });
  }

  movimientos.sort((a, b) => {
    const fechaA = convertirFechaReporte(
      a.fecha_orden || a.fecha
    );
    const fechaB = convertirFechaReporte(
      b.fecha_orden || b.fecha
    );

    return (
      (fechaA ? fechaA.getTime() : 0) -
      (fechaB ? fechaB.getTime() : 0)
    );
  });

  let saldo = 0;
  movimientos.forEach((movimiento) => {
    saldo += movimiento.haber - movimiento.debe;
    movimiento.saldo = saldo;
    delete movimiento.fecha_orden;
  });

  return responderJSON({
    rut: colaborador.rut,
    colaborador: colaborador.colaborador,
    empresa: colaborador.empresa,
    centro_costo: colaborador.centro_costo,
    total_abonos: totalAbonos,
    total_rendido: totalRendido,
    saldo_actual: totalAbonos - totalRendido,
    cantidad_movimientos: movimientos.length,
    movimientos: movimientos
  });
}

function obtenerResumenFinanciero(
  hojaAbonos,
  hojaRespuestas,
  hojaColaboradores,
  permisos,
  empresaSeleccionada
) {
  const cacheKey = obtenerClaveCacheConsulta(
    "DASHBOARD",
    permisos,
    empresaSeleccionada
  );

  const cacheado = leerCacheJSON(cacheKey);
  if (cacheado && cacheado.fuente_financiera) {
    return responderJSON(cacheado);
  }

  if (!hojaAbonos || !hojaRespuestas) {
    return responderJSON({
      error:
        "No se encontraron las hojas ABONOS o RESPUESTAS."
    });
  }

  const empresasPorRut =
    obtenerEmpresasPorRut(
      hojaColaboradores
    );

  const puedeIncluirRut = (rut) => {
    const rutNormalizado =
      normalizarRutPermisos(rut);

    if (!rutNormalizado) {
      return false;
    }

    if (
      !esRolSuperAdmin(permisos) &&
      !usuarioTieneRutAsignado(
        permisos,
        rutNormalizado
      )
    ) {
      return false;
    }

    return empresaPermitidaParaConsulta(
      empresasPorRut[rutNormalizado] || "",
      permisos,
      empresaSeleccionada
    );
  };

  const porRut = {};

  const asegurarRut = (rut) => {
    const clave = normalizarRutPermisos(rut);

    if (!porRut[clave]) {
      porRut[clave] = {
        depositado: 0,
        presentado: 0,
        autorizado: 0,
        pendiente: 0,
        rechazado: 0
      };
    }

    return porRut[clave];
  };

  const datosAbonos = obtenerDatosHojaConsultaCached(
    hojaAbonos,
    "ABONOS"
  );

  if (datosAbonos.length > 1) {
    const encabezados = datosAbonos[0];
    const idxColaborador = buscarIndice(
      encabezados,
      "COLABORADOR"
    );
    const idxDeposito = buscarIndice(
      encabezados,
      "DEPOSITO"
    );
    const idxEstado = buscarIndice(
      encabezados,
      "ESTADO"
    );

    if (idxColaborador < 0 || idxDeposito < 0) {
      return responderJSON({
        error:
          "La hoja ABONOS debe contener COLABORADOR y DEPOSITO."
      });
    }

    datosAbonos.slice(1).forEach((fila) => {
      const estado =
        idxEstado >= 0
          ? String(
              valorFila(fila, idxEstado) ||
              "ACTIVO"
            ).trim().toUpperCase()
          : "ACTIVO";

      if (estado === "ANULADO") {
        return;
      }

      const rut = extraerRut(
        valorFila(fila, idxColaborador)
      );

      if (!puedeIncluirRut(rut)) {
        return;
      }

      asegurarRut(rut).depositado +=
        convertirNumero(
          valorFila(fila, idxDeposito)
        );
    });
  }

  const datosRespuestas = obtenerDatosHojaConsultaCached(
    hojaRespuestas,
    "RESPUESTAS"
  );

  if (datosRespuestas.length > 1) {
    const encabezados = datosRespuestas[0];
    const idxColaborador = buscarIndice(
      encabezados,
      "colaborador"
    );
    const idxRut = buscarIndice(
      encabezados,
      "rut"
    );
    const idxMonto = buscarIndiceCualquiera(
      encabezados,
      ["monto_total", "MONTO"]
    );
    const idxEstadoComprobante = buscarIndice(
      encabezados,
      "estado_comprobante"
    );
    const idxEstadoRendicion = buscarIndice(
      encabezados,
      "estado_rendicion"
    );

    if (idxColaborador < 0 || idxMonto < 0) {
      return responderJSON({
        error:
          "RESPUESTAS debe contener colaborador y monto_total."
      });
    }

    datosRespuestas.slice(1).forEach((fila) => {
      const colaborador = valorFila(
        fila,
        idxColaborador
      );
      const rut =
        extraerRut(colaborador) ||
        valorFila(fila, idxRut);

      if (!puedeIncluirRut(rut)) {
        return;
      }

      const monto = convertirNumero(
        valorFila(fila, idxMonto)
      );
      const acumulado = asegurarRut(rut);
      acumulado.presentado += monto;

      const estadoComprobante =
        idxEstadoComprobante >= 0
          ? String(
              valorFila(
                fila,
                idxEstadoComprobante
              ) || ""
            ).trim().toUpperCase()
          : "";
      const estadoRendicion =
        idxEstadoRendicion >= 0
          ? String(
              valorFila(
                fila,
                idxEstadoRendicion
              ) || "PENDIENTE"
            ).trim().toUpperCase()
          : "PENDIENTE";

      if (
        estadoComprobante === "AUTORIZADO" ||
        (
          !estadoComprobante &&
          estadoRendicion === "AUTORIZADA"
        )
      ) {
        acumulado.autorizado += monto;
      } else if (
        estadoComprobante === "RECHAZADO" ||
        estadoRendicion === "RECHAZADA"
      ) {
        acumulado.rechazado += monto;
      } else {
        acumulado.pendiente += monto;
      }
    });
  }

  let montoDepositado = 0;
  let montoPresentado = 0;
  let montoAutorizado = 0;
  let montoPendienteRevision = 0;
  let montoRechazado = 0;
  let montoPorRendir = 0;
  let montoFavorColaboradores = 0;
  let colaboradoresConSaldo = 0;
  let colaboradoresSaldoFavor = 0;

  Object.keys(porRut).forEach((rut) => {
    const item = porRut[rut];
    montoDepositado += item.depositado;
    montoPresentado += item.presentado;
    montoAutorizado += item.autorizado;
    montoPendienteRevision += item.pendiente;
    montoRechazado += item.rechazado;

    const saldo =
      item.depositado - item.autorizado;

    if (saldo > 0) {
      montoPorRendir += saldo;
      colaboradoresConSaldo++;
    } else if (saldo < 0) {
      montoFavorColaboradores +=
        Math.abs(saldo);
      colaboradoresSaldoFavor++;
    }
  });

  const resultado = {
    fuente_financiera:
      "ABONOS_RESPUESTAS",
    monto_depositado:
      montoDepositado,
    monto_presentado:
      montoPresentado,
    monto_autorizado:
      montoAutorizado,
    monto_rendido:
      montoAutorizado,
    monto_pendiente_revision:
      montoPendienteRevision,
    monto_rechazado:
      montoRechazado,
    monto_por_rendir:
      montoPorRendir,
    colaboradores_con_saldo:
      colaboradoresConSaldo,
    monto_favor_colaboradores:
      montoFavorColaboradores,
    colaboradores_saldo_favor:
      colaboradoresSaldoFavor
  };

  guardarCacheJSON(cacheKey, resultado, 30);
  return responderJSON(resultado);
}


// ======================================================
// CONVERSIÓN DE DINERO
// ======================================================

function convertirNumero(valor) {
  if (
    valor === null ||
    valor === undefined ||
    valor === ""
  ) {
    return 0;
  }

  let texto =
    String(valor).trim();

  const esNegativo =
    texto.includes("-") ||
    texto.includes("(");

  texto = texto
    .replace(/[^0-9,.-]/g, "")
    .replace(/-/g, "")
    .trim();

  const ultimoPunto =
    texto.lastIndexOf(".");
  const ultimaComa =
    texto.lastIndexOf(",");

  if (
    ultimoPunto >= 0 &&
    ultimaComa >= 0
  ) {
    const separadorDecimal =
      ultimoPunto > ultimaComa
        ? "."
        : ",";

    const separadorMiles =
      separadorDecimal === "."
        ? /,/g
        : /\./g;

    texto = texto.replace(
      separadorMiles,
      ""
    );

    if (separadorDecimal === ",") {
      texto = texto.replace(",", ".");
    }
  } else if (ultimaComa >= 0) {
    const decimales =
      texto.length - ultimaComa - 1;

    texto = decimales === 3
      ? texto.replace(/,/g, "")
      : texto.replace(",", ".");
  } else if (ultimoPunto >= 0) {
    const decimales =
      texto.length - ultimoPunto - 1;

    if (decimales === 3) {
      texto = texto.replace(/\./g, "");
    }
  }

  const numero =
    Number(texto) || 0;

  return esNegativo
    ? -numero
    : numero;
}


// ======================================================
// RESPUESTA JSON
// ======================================================

function responderJSON(datos) {
  const callback = String(
    CALLBACK_JSONP || ""
  ).trim();

  const callbackValido =
    /^[A-Za-z_$][0-9A-Za-z_$]*$/.test(
      callback
    );

  if (callbackValido) {
    return ContentService
      .createTextOutput(
        callback + "(" +
        JSON.stringify(datos) +
        ");"
      )
      .setMimeType(
        ContentService.MimeType.JAVASCRIPT
      );
  }

  return ContentService
    .createTextOutput(
      JSON.stringify(datos)
    )
    .setMimeType(
      ContentService.MimeType.JSON
    );
}

// ======================================================

function obtenerFotoDriveComoBase64(
  hojaRespuestas,
  idArchivo
) {
  try {
    const idLimpio =
      String(
        idArchivo || ""
      ).trim();

    if (!idLimpio) {
      return responderJSON({
        error:
          "No se indicó el ID del archivo de Drive."
      });
    }

    if (
      !archivoDrivePerteneceARendiciones(
        hojaRespuestas,
        idLimpio
      )
    ) {
      return responderJSON({
        error:
          "El archivo solicitado no está asociado a una rendición."
      });
    }

    const archivo =
      DriveApp.getFileById(
        idLimpio
      );

    const blob =
      archivo.getBlob();

    const mimeType =
      blob.getContentType() ||
      "application/octet-stream";

    const base64 =
      Utilities.base64Encode(
        blob.getBytes()
      );

    return responderJSON({
      id: idLimpio,
      nombre:
        archivo.getName(),
      mimeType:
        mimeType,
      dataUrl:
        "data:" +
        mimeType +
        ";base64," +
        base64
    });

  } catch (error) {
    return responderJSON({
      error:
        "No fue posible obtener la fotografía desde Google Drive.",
      detalle:
        String(
          error &&
          error.message
            ? error.message
            : error
        )
    });
  }
}


function archivoDrivePerteneceARendiciones(
  hoja,
  idArchivo
) {
  if (!hoja || !idArchivo) {
    return false;
  }

  const ultimaFila =
    hoja.getLastRow();

  const ultimaColumna =
    hoja.getLastColumn();

  if (ultimaFila < 2) {
    return false;
  }

  const encabezados = hoja
    .getRange(
      1,
      1,
      1,
      ultimaColumna
    )
    .getDisplayValues()[0];

  const idxFotografia =
    buscarIndice(
      encabezados,
      "FOTOGRAFIA"
    );

  if (idxFotografia === -1) {
    return false;
  }

  const valores = hoja
    .getRange(
      2,
      idxFotografia + 1,
      ultimaFila - 1,
      1
    )
    .getDisplayValues();

  for (
    let i = 0;
    i < valores.length;
    i++
  ) {
    const enlace =
      String(
        valores[i][0] || ""
      ).trim();

    if (!enlace) {
      continue;
    }

    const idEncontrado =
      extraerIdDrive(
        enlace
      );

    if (
      idEncontrado ===
      idArchivo
    ) {
      return true;
    }
  }

  return false;
}


function extraerIdDrive(valor) {
  const texto =
    String(
      valor || ""
    ).trim();

  if (!texto) {
    return "";
  }

  const marcador =
    "/file/d/";

  const posicion =
    texto.indexOf(
      marcador
    );

  if (posicion >= 0) {
    const resto =
      texto.substring(
        posicion +
        marcador.length
      );

    return resto
      .split("/")[0]
      .split("?")[0];
  }

  const coincidenciaParametro =
    texto.match(
      /[?&]id=([a-zA-Z0-9_-]+)/
    );

  if (
    coincidenciaParametro &&
    coincidenciaParametro[1]
  ) {
    return coincidenciaParametro[1];
  }

  if (
    /^[a-zA-Z0-9_-]+$/.test(
      texto
    )
  ) {
    return texto;
  }

  return "";
}

function obtenerPermisosUsuario(
  emailUsuario
) {
  const emailBuscado = String(emailUsuario || "").trim().toLowerCase();

  if (!emailBuscado) {
    throw new Error(
      "Debe indicar un correo de usuario."
    );
  }

  const digest = Utilities.computeDigest(
    Utilities.DigestAlgorithm.MD5,
    emailBuscado
  );
  const cacheKey =
    "PERM_" +
    Utilities.base64EncodeWebSafe(digest).replace(/=+$/g, "");

  const cacheado = leerCacheJSON(cacheKey);
  if (cacheado) {
    return cacheado;
  }

  const libro =
    obtenerLibroRendiciones();

  const hojaUsuarios =
    libro.getSheetByName(
      "USUARIOS"
    );

  const hojaAsignaciones =
    libro.getSheetByName(
      "USUARIO_COLABORADORES"
    );

  if (
    !hojaUsuarios ||
    !hojaAsignaciones
  ) {
    throw new Error(
      "No se encontraron las hojas USUARIOS o USUARIO_COLABORADORES."
    );
  }

  // ======================================================
  // BUSCAR USUARIO
  // ======================================================

  const datosUsuarios =
    hojaUsuarios
      .getDataRange()
      .getDisplayValues();

  if (datosUsuarios.length < 2) {
    return {
      encontrado: false,
      email: emailBuscado
    };
  }

  const encabezadosUsuarios =
    datosUsuarios[0].map(
      (valor) =>
        String(valor || "")
          .trim()
          .toLowerCase()
    );

  const idxEmail =
    encabezadosUsuarios.indexOf(
      "email"
    );

  const idxNombre =
    encabezadosUsuarios.indexOf(
      "nombre"
    );

  const idxEmpresa =
    encabezadosUsuarios.indexOf(
      "empresa"
    );

  const idxRol =
    encabezadosUsuarios.indexOf(
      "rol"
    );

  const idxActivo =
    encabezadosUsuarios.indexOf(
      "activo"
    );

  if (
    idxEmail < 0 ||
    idxNombre < 0 ||
    idxEmpresa < 0 ||
    idxRol < 0 ||
    idxActivo < 0
  ) {
    throw new Error(
      "La hoja USUARIOS no tiene todos los encabezados requeridos."
    );
  }

  let usuarioEncontrado =
    null;

  for (
    let i = 1;
    i < datosUsuarios.length;
    i++
  ) {
    const fila =
      datosUsuarios[i];

    const emailFila =
      String(
        fila[idxEmail] || ""
      )
        .trim()
        .toLowerCase();

    if (
      emailFila ===
      emailBuscado
    ) {
      usuarioEncontrado = {
        email:
          emailFila,

        nombre:
          String(
            fila[idxNombre] || ""
          ).trim(),

        empresa:
          String(
            fila[idxEmpresa] || ""
          ).trim(),

        rol:
          String(
            fila[idxRol] || ""
          )
            .trim()
            .toUpperCase(),

        activo:
          String(
            fila[idxActivo] || ""
          )
            .trim()
            .toUpperCase() === "SI"
      };

      break;
    }
  }

  if (!usuarioEncontrado) {
    return {
      encontrado: false,
      email: emailBuscado
    };
  }

  // ======================================================
  // BUSCAR COLABORADORES ASIGNADOS
  // ======================================================

  const datosAsignaciones =
    hojaAsignaciones
      .getDataRange()
      .getDisplayValues();

  if (datosAsignaciones.length === 0) {
    return {
      encontrado: true,
      email:
        usuarioEncontrado.email,
      nombre:
        usuarioEncontrado.nombre,
      empresa:
        usuarioEncontrado.empresa,
      rol:
        usuarioEncontrado.rol,
      activo:
        usuarioEncontrado.activo,
      colaboradores: []
    };
  }

  const encabezadosAsignaciones =
    datosAsignaciones[0].map(
      (valor) =>
        String(valor || "")
          .trim()
          .toLowerCase()
    );

  const idxEmailUsuario =
    encabezadosAsignaciones.indexOf(
      "email_usuario"
    );

  const idxRutColaborador =
    encabezadosAsignaciones.indexOf(
      "rut_colaborador"
    );

  if (
    idxEmailUsuario < 0 ||
    idxRutColaborador < 0
  ) {
    throw new Error(
      "La hoja USUARIO_COLABORADORES no tiene los encabezados requeridos."
    );
  }

  const colaboradores = [];

  for (
    let i = 1;
    i < datosAsignaciones.length;
    i++
  ) {
    const fila =
      datosAsignaciones[i];

    const emailFila =
      String(
        fila[idxEmailUsuario] || ""
      )
        .trim()
        .toLowerCase();

    const rutFila =
      String(
        fila[idxRutColaborador] || ""
      ).trim();

    if (
      emailFila === emailBuscado &&
      rutFila &&
      !colaboradores.includes(
        rutFila
      )
    ) {
      colaboradores.push(
        rutFila
      );
    }
  }

  const permisos = {
    encontrado: true,
    email:
      usuarioEncontrado.email,
    nombre:
      usuarioEncontrado.nombre,
    empresa:
      usuarioEncontrado.empresa,
    rol:
      usuarioEncontrado.rol,
    activo:
      usuarioEncontrado.activo,
    colaboradores:
      colaboradores
  };

  guardarCacheJSON(cacheKey, permisos, 30);
  return permisos;
}
